### 2.1 分析进程内存映射

```c
// 文件: exp2_process_maps.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/mman.h>

// 打印当前进程的内存映射
void print_memory_maps() {
    char buf[256];
    FILE *fp = fopen("/proc/self/maps", "r");

    printf("进程 %d 的内存映射:\n", getpid());
    printf("================================================================\n");
    printf("%-18s %-5s %-5s %-10s %-10s %s\n",
           "地址范围", "权限", "共享", "偏移", "设备", "路径");
    printf("----------------------------------------------------------------\n");

    while (fgets(buf, sizeof(buf), fp)) {
        printf("%s", buf);
    }
    fclose(fp);
}

int main() {
    printf("========== 初始状态 ==========\n");
    print_memory_maps();

    // 分配堆内存
    printf("\n========== malloc后 ==========\n");
    void *heap = malloc(1024 * 1024);  // 1MB
    print_memory_maps();

    // 创建匿名映射
    printf("\n========== mmap匿名映射后 ==========\n");
    void *anon = mmap(NULL, 2 * 1024 * 1024,  // 2MB
                      PROT_READ | PROT_WRITE,
                      MAP_PRIVATE | MAP_ANONYMOUS,
                      -1, 0);
    print_memory_maps();

    // 创建共享映射
    printf("\n========== mmap共享映射后 ==========\n");
    void *shared = mmap(NULL, 1024 * 1024,
                        PROT_READ | PROT_WRITE,
                        MAP_SHARED | MAP_ANONYMOUS,
                        -1, 0);
    print_memory_maps();

    // 清理
    free(heap);
    munmap(anon, 2 * 1024 * 1024);
    munmap(shared, 1024 * 1024);

    printf("\n按Enter继续...");
    getchar();
    return 0;
}
```

### 2.2 VMA分析脚本

```python
#!/usr/bin/env python3
# 文件: exp2_vma_analyzer.py
# 分析进程VMA统计

import sys
from collections import defaultdict

def analyze_vma(pid):
    try:
        with open(f'/proc/{pid}/maps', 'r') as f:
            lines = f.readlines()
    except:
        print(f"无法读取进程 {pid} 的内存映射")
        return

    stats = defaultdict(lambda: {'count': 0, 'size': 0})

    for line in lines:
        parts = line.split()
        addr_range = parts[0]
        perms = parts[1]

        start, end = [int(x, 16) for x in addr_range.split('-')]
        size = end - start

        # 分类
        if len(parts) > 5:
            path = parts[5]
            if '[' in path:
                category = path
            else:
                category = '文件映射'
        elif 'r-xp' in perms:
            category = '代码段'
        elif 'rw-p' in perms:
            category = '数据段'
        else:
            category = '其他'

        stats[category]['count'] += 1
        stats[category]['size'] += size

    print(f"\n进程 {pid} VMA统计:")
    print("=" * 50)
    for cat, data in sorted(stats.items(), key=lambda x: -x[1]['size']):
        size_mb = data['size'] / (1024 * 1024)
        print(f"{cat:20s}: {data['count']:4d} 个, {size_mb:8.2f} MB")

if __name__ == '__main__':
    pid = sys.argv[1] if len(sys.argv) > 1 else str(os.getpid())
    analyze_vma(pid)
```

## 🧪 实验三：缺页中断追踪

> [!tip] 目标
> 观察页面错误处理过程

### 3.1 统计缺页次数

```c
// 文件: exp3_page_fault.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/mman.h>
#include <sys/resource.h>

// 获取缺页次数
void get_page_faults(long *minor, long *major) {
    struct rusage usage;
    getrusage(RUSAGE_SELF, &usage);
    *minor = usage.ru_minflt;
    *major = usage.ru_majflt;
}

int main() {
    long minor_before, major_before, minor_after, major_after;

    get_page_faults(&minor_before, &major_before);

    // 触发缺页的几种方式

    // 方式1: 访问malloc的内存（首次访问）
    printf("方式1: 访问malloc内存\n");
    char *heap = malloc(10 * 1024 * 1024);  // 10MB
    for (int i = 0; i < 10 * 1024 * 1024; i += 4096) {
        heap[i] = 'x';  // 触发缺页
    }

    get_page_faults(&minor_after, &major_after);
    printf("  Minor faults: %ld\n", minor_after - minor_before);
    printf("  Major faults: %ld\n", major_after - major_before);

    minor_before = minor_after;
    major_before = major_after;

    // 方式2: mmap匿名映射
    printf("\n方式2: mmap匿名映射\n");
    char *mapped = mmap(NULL, 10 * 1024 * 1024,
                        PROT_READ | PROT_WRITE,
                        MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
    for (int i = 0; i < 10 * 1024 * 1024; i += 4096) {
        mapped[i] = 'x';
    }

    get_page_faults(&minor_after, &major_after);
    printf("  Minor faults: %ld\n", minor_after - minor_before);
    printf("  Major faults: %ld\n", major_after - major_before);

    minor_before = minor_after;
    major_before = major_after;

    // 方式3: 文件映射
    printf("\n方式3: 文件映射\n");
    FILE *fp = fopen("/bin/ls", "r");
    fseek(fp, 0, SEEK_END);
    long fsize = ftell(fp);
    fseek(fp, 0, SEEK_SET);

    char *file_map = mmap(NULL, fsize, PROT_READ, MAP_PRIVATE, fileno(fp), 0);
    volatile char sum = 0;
    for (long i = 0; i < fsize; i += 4096) {
        sum += file_map[i];
    }

    get_page_faults(&minor_after, &major_after);
    printf("  Minor faults: %ld\n", minor_after - minor_before);
    printf("  Major faults: %ld\n", major_after - major_before);

    // 清理
    free(heap);
    munmap(mapped, 10 * 1024 * 1024);
    munmap(file_map, fsize);
    fclose(fp);

    return 0;
}
```

### 4.2 内存碎片化测试

```c
// 文件: exp4_fragmentation.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/mman.h>

#define NUM_BLOCKS 1000
#define BLOCK_SIZE (64 * 1024)  // 64KB

int main() {
    void *blocks[NUM_BLOCKS];

    printf("分配 %d 个 %dKB 内存块...\n", NUM_BLOCKS, BLOCK_SIZE/1024);

    // 分配
    for (int i = 0; i < NUM_BLOCKS; i++) {
        blocks[i] = malloc(BLOCK_SIZE);
        if (!blocks[i]) {
            printf("分配失败在第 %d 块\n", i);
            break;
        }
        // 触发页面分配
        memset(blocks[i], 'x', BLOCK_SIZE);
    }

    printf("分配完成，按Enter继续...\n");
    getchar();

    // 间隔释放，制造碎片
    printf("间隔释放，制造碎片...\n");
    for (int i = 0; i < NUM_BLOCKS; i += 2) {
        free(blocks[i]);
        blocks[i] = NULL;
    }

    printf("碎片化后，按Enter继续...\n");
    getchar();

    // 尝试分配大块连续内存
    printf("尝试分配 1MB 连续内存...\n");
    void *large = malloc(1024 * 1024);
    if (large) {
        printf("成功分配\n");
        free(large);
    } else {
        printf("分配失败（碎片影响）\n");
    }

    // 清理
    for (int i = 0; i < NUM_BLOCKS; i++) {
        if (blocks[i]) free(blocks[i]);
    }

    return 0;
}
```

### 5.2 自定义内核模块：Slab统计

```c
// 文件: exp5_slab_module/slab_stat.c
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/slab.h>
#include <linux/mm.h>

#define MY_CACHE_NAME "my_test_cache"
#define MY_OBJECT_SIZE 256

static struct kmem_cache *my_cache;

static int __init slab_stat_init(void)
{
    void *obj1, *obj2, *obj3;

    pr_info("Slab统计模块加载\n");

    // 创建自定义缓存
    my_cache = kmem_cache_create(MY_CACHE_NAME,
                                 MY_OBJECT_SIZE, 0,
                                 SLAB_HWCACHE_ALIGN, NULL);
    if (!my_cache) {
        pr_err("创建缓存失败\n");
        return -ENOMEM;
    }

    pr_info("创建缓存: %s, 对象大小: %d\n",
            MY_CACHE_NAME, MY_OBJECT_SIZE);

    // 分配对象
    obj1 = kmem_cache_alloc(my_cache, GFP_KERNEL);
    obj2 = kmem_cache_alloc(my_cache, GFP_KERNEL);
    obj3 = kmem_cache_alloc(my_cache, GFP_KERNEL);

    pr_info("分配了3个对象\n");
    pr_info("查看: cat /proc/slabinfo | grep %s\n", MY_CACHE_NAME);

    // 释放对象（保留一个）
    kmem_cache_free(my_cache, obj1);
    kmem_cache_free(my_cache, obj2);
    pr_info("释放了2个对象，保留1个\n");

    // 不释放obj3，留着观察

    return 0;
}

static void __exit slab_stat_exit(void)
{
    pr_info("Slab统计模块卸载\n");
    kmem_cache_destroy(my_cache);
}

module_init(slab_stat_init);
module_exit(slab_stat_exit);
MODULE_LICENSE("GPL");
```

    echo "$timestamp,$pgfault,$pgmajfault,$pgscan,$pgsteal,$kswapd" >> reclaim_log.txt
    sleep 1
done

## 🧪 实验七：写时复制（COW）

> [!tip] 目标
>
**写时复制（Copy-on-Write，简称 COW）** 是一种计算机程序设计领域的优化策略。
其核心思想是：当多个调用者（caller）请求相同的资源时，初始状态下它们会共享同一份资源，并不会真的复制。只有当某个调用者试图修改这份共享资源时，系统才会真正为其创建一个私有的副本，以供其修改。

---

### 核心思想：懒惰复制

写时复制的精髓在于**推迟**复制操作。它假设大部分资源在共享期间不会被修改，因此通过延迟复制来避免不必要的开销。

- **读操作**：多个访问者可以安全地共享同一份数据，无需复制。
    
- **写操作**：当某个访问者试图修改数据时，系统会拦截这个操作，先为这个访问者创建一个副本，然后将修改操作应用到这个新副本上。这样，其他访问者看到的仍然是原来的数据，不受影响。
### 7.1 COW验证程序

```c
// 文件: exp7_cow.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/wait.h>
#include <sys/resource.h>

void print_memory_usage(const char *phase) {
    FILE *fp = fopen("/proc/self/status", "r");
    char line[256];

    printf("=== %s ===\n", phase);
    while (fgets(line, sizeof(line), fp)) {
        if (strstr(line, "VmRSS") || strstr(line, "VmSize") ||
            strstr(line, "VmData") || strstr(line, "VmStk")) {
            printf("%s", line);
        }
    }
    fclose(fp);
}

int main() {
    // 分配共享数据
    size_t size = 100 * 1024 * 1024;  // 100MB
    char *shared_data = malloc(size);

    // 初始化数据
    memset(shared_data, 'A', size);

    print_memory_usage("fork前");

    pid_t pid = fork();

    if (pid == 0) {
        // 子进程
        print_memory_usage("子进程fork后（修改前）");

        // 修改部分数据，触发COW
        printf("子进程: 修改数据...\n");
        for (size_t i = 0; i < size; i += 4096) {
            shared_data[i] = 'B';  // 每页一个字节
        }

        print_memory_usage("子进程修改后（COW发生）");

        free(shared_data);
        exit(0);
    } else {
        // 父进程
        wait(NULL);

        print_memory_usage("父进程（子进程退出后）");

        free(shared_data);
    }

    return 0;
}
```
