## 🧪 实验一：进程状态观察

> [!tip] 目标
> 理解进程状态转换、进程关系

### 1.1 观察进程状态

```c
// 文件: exp1_process_state.c
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>

int main() {
    pid_t pid;

    printf("父进程 PID=%d\n", getpid());
    printf("观察进程状态: watch -n 1 'ps -eo pid,ppid,stat,cmd | grep exp1'\n\n");

    pid = fork();

    if (pid == 0) {
        // 子进程
        printf("子进程 PID=%d, 开始睡眠...\n", getpid());
        sleep(30);
        printf("子进程退出\n");
        exit(0);
    }

    printf("子进程 PID=%d 已创建\n", pid);
    printf("按Enter让子进程成为孤儿进程...\n");
    getchar();

    // 父进程退出，子进程成为孤儿
    printf("父进程退出，子进程成为孤儿\n");
    return 0;
}
```

## 🧪 实验二：进程调度分析

> [!tip] 目标
> 理解进程优先级、调度策略

### 2.1 调度策略与优先级

```c
// 文件: exp2_sched_info.c
#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sched.h>
#include <sys/resource.h>
#include <sys/syscall.h>

const char *policy_name(int policy) {
    switch (policy) {
        case SCHED_OTHER: return "SCHED_OTHER";
        case SCHED_FIFO:  return "SCHED_FIFO";
        case SCHED_RR:    return "SCHED_RR";
        case SCHED_BATCH: return "SCHED_BATCH";
        case SCHED_IDLE:  return "SCHED_IDLE";
        default: return "UNKNOWN";
    }
}

int main() {
    int policy;
    struct sched_param param;

    // 获取当前调度策略
    policy = sched_getscheduler(0);
    printf("当前调度策略: %s\n", policy_name(policy));

    // 获取优先级
    int prio = getpriority(PRIO_PROCESS, 0);
    printf("Nice值: %d\n", prio);

    // 获取调度参数
    sched_getparam(0, &param);
    printf("调度优先级: %d\n", param.sched_priority);

    // 获取CPU亲和性
    cpu_set_t mask;
    CPU_ZERO(&mask);
    sched_getaffinity(0, sizeof(mask), &mask);

    printf("可运行的CPU: ");
    for (int i = 0; i < CPU_SETSIZE; i++) {
        if (CPU_ISSET(i, &mask)) {
            printf("%d ", i);
        }
    }
    printf("\n");

    return 0;
}
```

### 2.2 CPU绑定实验

```c
// 文件: exp2_cpu_affinity.c
#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sched.h>
#include <pthread.h>

void *worker(void *arg) {
    int cpu = *(int *)arg;
    printf("线程绑定到CPU %d\n", cpu);

    // 忙循环
    while (1) {
        // CPU密集任务
        for (volatile int i = 0; i < 1000000; i++);
    }
    return NULL;
}

int main() {
    pthread_t threads[4];
    int cpus[4] = {0, 1, 2, 3};
    cpu_set_t cpuset;

    for (int i = 0; i < 4; i++) {
        pthread_create(&threads[i], NULL, worker, &cpus[i]);

        // 绑定线程到特定CPU
        CPU_ZERO(&cpuset);
        CPU_SET(i, &cpuset);
        pthread_setaffinity_np(threads[i], sizeof(cpuset), &cpuset);
    }

    printf("运行中... 使用 top -H 查看线程\n");
    printf("按Ctrl+C退出\n");

    for (int i = 0; i < 4; i++) {
        pthread_join(threads[i], NULL);
    }

    return 0;
}
```

### 2.3 调度延迟测量

```c
// 文件: exp2_sched_latency.c
#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <time.h>
#include <sched.h>

int main() {
    struct timespec start, end;
    long long total_ns = 0;
    int iterations = 1000;

    printf("测量调度延迟...\n");

    for (int i = 0; i < iterations; i++) {
        clock_gettime(CLOCK_MONOTONIC, &start);

        // 主动让出CPU
        sched_yield();

        clock_gettime(CLOCK_MONOTONIC, &end);

        long long diff = (end.tv_sec - start.tv_sec) * 1000000000LL +
                         (end.tv_nsec - start.tv_nsec);
        total_ns += diff;
    }

    printf("平均调度延迟: %lld ns\n", total_ns / iterations);

    return 0;
}
```

## 🧪 实验三：上下文切换分析

> [!tip] 目标
> 测量上下文切换开销

### 3.1 管道通信测量

```c
// 文件: exp3_context_switch.c
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <time.h>
#include <sys/wait.h>

int main() {
    int pipe1[2], pipe2[2];
    char buf[1] = {'x'};
    struct timespec start, end;

    pipe(pipe1);
    pipe(pipe2);

    pid_t pid = fork();

    if (pid == 0) {
        // 子进程
        for (int i = 0; i < 10000; i++) {
            read(pipe1[0], buf, 1);
            write(pipe2[1], buf, 1);
        }
        exit(0);
    }

    // 父进程
    clock_gettime(CLOCK_MONOTONIC, &start);

    for (int i = 0; i < 10000; i++) {
        write(pipe1[1], buf, 1);
        read(pipe2[0], buf, 1);
    }

    clock_gettime(CLOCK_MONOTONIC, &end);

    wait(NULL);

    long long diff = (end.tv_sec - start.tv_sec) * 1000000000LL +
                     (end.tv_nsec - start.tv_nsec);

    printf("10000次上下文切换耗时: %lld ns\n", diff);
    printf("平均每次: %lld ns\n", diff / 10000);

    close(pipe1[0]); close(pipe1[1]);
    close(pipe2[0]); close(pipe2[1]);

    return 0;
}
```

## 🧪 实验四：中断统计与分析

> [!tip] 目标
> 理解中断分布和中断负载

### 4.1 中断监控脚本

```bash
#!/bin/bash
# 文件: exp4_irq_monitor.sh

LOG_FILE="irq_log.txt"

echo "时间,中断总数,时钟中断,键盘中断,网卡中断" > $LOG_FILE

for i in {1..60}; do
    timestamp=$(date +%H:%M:%S)

    # 获取中断统计
    stats=$(cat /proc/interrupts | awk '
        NR==1 {next}
        {
            for(i=2; i<=NF-1; i++) {
                total[i-2] += $i
            }
            gsub(/[^0-9]/, "", $NF)
            if ($NF ~ /timer/) timer = $2
            if ($NF ~ /keyboard/ || $NF ~ /i8042/) keyboard = $2
            if ($NF ~ /eth/ || $NF ~ /enp/) network = $2
        }
        END {
            sum = 0
            for(i in total) sum += total[i]
            print sum, timer+0, keyboard+0, network+0
        }
    ')

    echo "$timestamp,$stats" >> $LOG_FILE
    sleep 1
done

echo "数据已保存到 $LOG_FILE"
```

## 🧪 实验五：内核模块中断处理

> [!tip] 目标

### 5.1 共享中断模块

```c
// 文件: exp5_irq_module/irq_test.c
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/interrupt.h>
#include <linux/proc_fs.h>
#include <linux/seq_file.h>

#define IRQ_NUM 1  // 键盘中断

static int irq_count = 0;
static ktime_t last_irq_time;

static irqreturn_t my_irq_handler(int irq, void *dev_id)
{
    irq_count++;
    last_irq_time = ktime_get();

    return IRQ_HANDLED;
}

static int irq_proc_show(struct seq_file *m, void *v)
{
    seq_printf(m, "中断次数: %d\n", irq_count);
    seq_printf(m, "最后中断时间: %lld ns\n", ktime_to_ns(last_irq_time));
    return 0;
}

static int irq_proc_open(struct inode *inode, struct file *file)
{
    return single_open(file, irq_proc_show, NULL);
}

static const struct proc_ops irq_proc_ops = {
    .proc_open = irq_proc_open,
    .proc_read = seq_read,
    .proc_lseek = seq_lseek,
    .proc_release = single_release,
};

static int __init irq_test_init(void)
{
    int ret;

    ret = request_irq(IRQ_NUM, my_irq_handler,
                      IRQF_SHARED, "irq_test", &irq_count);
    if (ret) {
        pr_err("注册中断失败: %d\n", ret);
        return ret;
    }

    proc_create("irq_test", 0, NULL, &irq_proc_ops);

    pr_info("中断测试模块加载，IRQ=%d\n", IRQ_NUM);
    return 0;
}

static void __exit irq_test_exit(void)
{
    remove_proc_entry("irq_test", NULL);
    free_irq(IRQ_NUM, &irq_count);
    pr_info("中断测试模块卸载\n");
}

module_init(irq_test_init);
module_exit(irq_test_exit);
MODULE_LICENSE("GPL");
```

### 5.2 Tasklet模块

// 文件: exp5_tasklet_module/tasklet_test.c
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/interrupt.h>
#include <linux/proc_fs.h>
#include <linux/seq_file.h>

static int tasklet_count = 0;

static void my_tasklet_func(unsigned long data)
{
    tasklet_count++;
    pr_info("Tasklet执行，计数: %d\n", tasklet_count);
}

static DECLARE_TASKLET(my_tasklet, my_tasklet_func, 0);

static int tasklet_proc_show(struct seq_file *m, void *v)
{
    seq_printf(m, "Tasklet执行次数: %d\n", tasklet_count);
    seq_printf(m, "Tasklet状态: %lu\n", my_tasklet.state);

static int tasklet_proc_open(struct inode *inode, struct file *file)
{
    return single_open(file, tasklet_proc_show, NULL);
}

static ssize_t tasklet_proc_write(struct file *file, const char __user *buf,
                                   size_t count, loff_t *ppos)
{
    tasklet_schedule(&my_tasklet);
    return count;
}

static const struct proc_ops tasklet_proc_ops = {
    .proc_open = tasklet_proc_open,
    .proc_read = seq_read,
    .proc_write = tasklet_proc_write,
    .proc_lseek = seq_lseek,
    .proc_release = single_release,

{
    proc_create("tasklet_test", 0666, NULL, &tasklet_proc_ops);
    pr_info("Tasklet测试模块加载\n");
    return 0;
}

{
    tasklet_kill(&my_tasklet);
    remove_proc_entry("tasklet_test", NULL);
    pr_info("Tasklet测试模块卸载\n");

module_init(tasklet_test_init);

### 6.1 工作队列模块

```c
// 文件: exp6_workqueue_module/workqueue_test.c
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/workqueue.h>
#include <linux/slab.h>
#include <linux/proc_fs.h>
#include <linux/seq_file.h>

static struct workqueue_struct *my_wq;
static struct delayed_work my_delayed_work;
static int work_count = 0;

static void my_work_handler(struct work_struct *work)
{
    work_count++;
    pr_info("工作队列执行，计数: %d，进程: %s (PID: %d)\n",
            work_count, current->comm, current->pid);

    // 可以睡眠
    msleep(100);
}

static int work_proc_show(struct seq_file *m, void *v)
{
    seq_printf(m, "工作执行次数: %d\n", work_count);
    return 0;
}

static int work_proc_open(struct inode *inode, struct file *file)
{
    return single_open(file, work_proc_show, NULL);
}

static ssize_t work_proc_write(struct file *file, const char __user *buf,
                                size_t count, loff_t *ppos)
{
    // 立即调度
    queue_work(my_wq, &my_delayed_work.work);

    // 延迟调度（1秒后）
    // queue_delayed_work(my_wq, &my_delayed_work, HZ);

    return count;
}

static const struct proc_ops work_proc_ops = {
    .proc_open = work_proc_open,
    .proc_read = seq_read,
    .proc_write = work_proc_write,
    .proc_lseek = seq_lseek,
    .proc_release = single_release,
};

static int __init workqueue_test_init(void)
{
    my_wq = create_workqueue("my_workqueue");
    if (!my_wq)
        return -ENOMEM;

    INIT_DELAYED_WORK(&my_delayed_work, my_work_handler);

    proc_create("workqueue_test", 0666, NULL, &work_proc_ops);
    pr_info("工作队列测试模块加载\n");
    return 0;
}

static void __exit workqueue_test_exit(void)
{
    cancel_delayed_work_sync(&my_delayed_work);
    flush_workqueue(my_wq);
    destroy_workqueue(my_wq);
    remove_proc_entry("workqueue_test", NULL);
    pr_info("工作队列测试模块卸载\n");
}

module_init(workqueue_test_init);
module_exit(workqueue_test_exit);
MODULE_LICENSE("GPL");
```
