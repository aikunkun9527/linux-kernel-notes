# 进程创建与销毁

> [!info] 目标
> 理解Linux进程的创建（fork/clone）和终止（exit）机制

## 进程创建概述

Linux进程创建主要涉及三个系统调用：
- **fork()**：创建子进程，复制父进程
- **vfork()**：创建子进程，共享父进程地址空间
- **clone()**：灵活的进程/线程创建接口

## fork系统调用

### 调用流程

```
用户态: fork()
    │
    ▼
内核态: sys_fork / sys_clone
    │
    ▼
kernel_clone()
    │
    ▼
copy_process()
    │
    ├── 分配task_struct
    ├── 分配PID
    ├── 复制/共享资源
    │   ├── copy_mm()
    │   ├── copy_fs()
    │   ├── copy_files()
    │   ├── copy_signal()
    │   └── copy_thread()
    │
    ▼
wake_up_new_task()
```

### copy_process() 详解

```c
// kernel/fork.c
static __latent_entropy struct task_struct *copy_process(
                    struct pid *pid, int trace, int node,
                    struct kernel_clone_args *args)
{
    struct task_struct *p;

    // 1. 分配task_struct
    p = alloc_task_struct(node);
    if (!p)
        return ERR_PTR(-ENOMEM);

    // 2. 复制父进程的task_struct
    memcpy(p, current, sizeof(*p));

    // 3. 初始化新进程
    p->pid = pid_nr(pid);
    p->start_time = ktime_get_ns();

    // 4. 复制各种资源
    retval = copy_mm(clone_flags, p);
    retval = copy_fs(clone_flags, p);
    retval = copy_files(clone_flags, p);
    retval = copy_signal(clone_flags, p);
    retval = copy_thread(clone_flags, args->stack, args->stack_size, p, args);

    // 5. 设置调度相关信息
    p->sched_class = current->sched_class;

    // 6. 加入运行队列
    wake_up_new_task(p);

    return p;
}
```

### 写时复制（COW）

fork()采用写时复制优化：

```
fork后初始状态：
┌─────────────────┐     ┌─────────────────┐
│ 父进程页表       │     │ 子进程页表       │
│ (只读)          │     │ (只读)          │
└────────┬────────┘     └────────┬────────┘
         │                       │
         └───────────┬───────────┘
                     ▼
              ┌─────────────┐
              │ 共享物理页面 │
              │ (标记只读)   │
              └─────────────┘

父进程写入时：
┌─────────────────┐
│ 父进程页表       │
│ (可写)          │
└────────┬────────┘
         │
         ▼
  ┌─────────────┐
  │ 新物理页面   │ ← 复制副本
  └─────────────┘

子进程页表指向原页面
```

### copy_mm() 实现

```c
static int copy_mm(unsigned long clone_flags, struct task_struct *tsk)
{
    struct mm_struct *mm, *oldmm;

    oldmm = current->mm;
    if (!oldmm)
        return 0;  // 内核线程

    // CLONE_VM表示共享地址空间（线程）
    if (clone_flags & CLONE_VM) {
        mm = oldmm;
        mmget(mm);
        tsk->mm = mm;
        tsk->active_mm = mm;
        return 0;
    }

    // 复制地址空间（进程）
    mm = dup_mm(tsk);
    if (!mm)
        return -ENOMEM;

    tsk->mm = mm;
    tsk->active_mm = mm;
    return 0;
}
```

### copy_thread() - 架构相关

```c
// arch/x86/kernel/process.c
int copy_thread(unsigned long clone_flags, unsigned long sp,
                unsigned long arg, struct task_struct *p,
                struct kernel_clone_args *args)
{
    struct pt_regs *childregs;

    // 获取子进程内核栈顶
    childregs = task_pt_regs(p);

    // 复制父进程的寄存器
    *childregs = *current_pt_regs();

    // 设置子进程返回值为0（fork返回0给子进程）
    childregs->ax = 0;

    // 设置子进程栈指针
    if (sp)
        childregs->sp = sp;

    // 设置子进程从ret_from_fork开始执行
    p->thread.ip = (unsigned long) ret_from_fork;

    return 0;
}
```

## clone系统调用

### clone标志

```c
// 主要clone标志
#define CSIGNAL         0x000000ff  // 信号掩码

#define CLONE_VM        0x00000100  // 共享地址空间
#define CLONE_FS        0x00000200  // 共享文件系统信息
#define CLONE_FILES     0x00000400  // 共享文件描述符表
#define CLONE_SIGHAND   0x00000800  // 共享信号处理

#define CLONE_PIDFD     0x00001000  // 创建pidfd
#define CLONE_PTRACE    0x00002000  // 被追踪
#define CLONE_VFORK     0x00004000  // 父进程等待子进程
#define CLONE_PARENT    0x00008000  // 与父进程共享父进程

#define CLONE_THREAD    0x00010000  // 加入同一线程组
#define CLONE_NEWNS     0x00020000  // 新mount命名空间
#define CLONE_SYSVSEM   0x00040000  // 共享SYSV信号量
#define CLONE_SETTLS    0x00080000  // 设置TLS
#define CLONE_PARENT_SETTID 0x00100000
#define CLONE_CHILD_CLEARTID 0x00200000
#define CLONE_DETACHED  0x00400000
#define CLONE_UNTRACED  0x00800000
#define CLONE_CHILD_SETTID 0x01000000
#define CLONE_NEWUTS    0x04000000  // 新UTS命名空间
#define CLONE_NEWIPC    0x08000000  // 新IPC命名空间
#define CLONE_NEWUSER   0x10000000  // 新用户命名空间
#define CLONE_NEWPID    0x20000000  // 新PID命名空间
#define CLONE_NEWNET    0x40000000  // 新网络命名空间
#define CLONE_IO        0x80000000  // 共享IO上下文
```

### clone与fork对比

```c
// fork - 复制进程
clone(SIGCHLD, 0);

// vfork - 父进程阻塞，子进程共享地址空间
clone(CLONE_VFORK | CLONE_VM | SIGCHLD, 0);

// pthread_create - 创建线程
clone(CLONE_VM | CLONE_FS | CLONE_FILES | CLONE_SIGHAND |
      CLONE_THREAD | CLONE_SYSVSEM, stack);
```

### 僵尸进程与孤儿进程

```
僵尸进程：
- 子进程已exit()，但父进程未wait()
- task_struct仍存在
- 占用PID资源
- 解决：父进程调用wait()

孤儿进程：
- 父进程先于子进程exit()
- 被init进程(PID 1)收养
- init负责回收资源
```

## 进程创建示例

### fork示例

```c
#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>

int main() {
    pid_t pid = fork();

    if (pid < 0) {
        perror("fork failed");
        return 1;
    } else if (pid == 0) {
        // 子进程
        printf("Child: PID=%d, PPID=%d\n", getpid(), getppid());
        execlp("/bin/ls", "ls", "-la", NULL);
    } else {
        // 父进程
        printf("Parent: PID=%d, Child PID=%d\n", getpid(), pid);
        wait(NULL);
        printf("Child exited\n");
    }

    return 0;
}
```

### clone创建线程

```c
#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sched.h>
#include <sys/wait.h>

#define STACK_SIZE (1024 * 1024)

static int thread_func(void *arg) {
    printf("Thread: PID=%d\n", getpid());
    return 0;
}

int main() {
    char *stack = malloc(STACK_SIZE);
    char *stack_top = stack + STACK_SIZE;

    pid_t tid = clone(thread_func, stack_top,
                      CLONE_VM | CLONE_FS | CLONE_FILES |
                      CLONE_SIGHAND | CLONE_THREAD,
                      NULL);

    if (tid == -1) {
        perror("clone failed");
        return 1;
    }

    printf("Main: PID=%d, TID=%d\n", getpid(), tid);
    sleep(1);
    return 0;
}
```
