# 进程地址空间

> [!info] 目标
> 理解Linux进程虚拟地址空间的组织与管理

## 地址空间概述

每个进程拥有独立的虚拟地址空间：
- 32位系统：4GB地址空间
- 64位系统：128TB用户空间 + 128TB内核空间

## 进程地址空间布局

### 32位布局

```
0xFFFFFFFF ┌──────────────────────┐
           │      内核空间         │ 1GB
0xC0000000 ├──────────────────────┤
           │      用户栈          │ ↓ 向下增长
           │         ↕           │
           ├──────────────────────┤
           │    内存映射区域       │ mmap区域
           │         ↕           │
           ├──────────────────────┤
           │       堆             │ ↑ 向上增长
0x08048000 ├──────────────────────┤
           │    数据段(.data)      │
           ├──────────────────────┤
           │    BSS段(.bss)        │
           ├──────────────────────┤
           │    代码段(.text)      │
           ├──────────────────────┤
           │    只读数据(.rodata)  │
0x00000000 └──────────────────────┘
```

## 核心数据结构

### mm_struct

```c
// include/linux/mm_types.h
struct mm_struct {
    // 内存区域链表和树
    struct vm_area_struct *mmap;    // VMA链表头
    struct rb_root mm_rb;           // VMA红黑树根
    unsigned long mmap_base;        // mmap区域起始地址
    unsigned long total_vm;         // 总虚拟内存页数
    unsigned long locked_vm;        // 锁定的内存页数

    // 代码段、数据段、堆、栈
    unsigned long start_code, end_code;  // 代码段范围
    unsigned long start_data, end_data;  // 数据段范围
    unsigned long start_brk, brk;        // 堆范围
    unsigned long start_stack;           // 栈起始地址
    unsigned long arg_start, arg_end;    // 命令行参数
    unsigned long env_start, env_end;    // 环境变量

    // 页表
    pgd_t *pgd;                     // 页全局目录

    // 引用计数
    atomic_t mm_users;              // 用户计数
    atomic_t mm_count;              // 引用计数

    // 进程统计
    unsigned long hiwater_rss;      // RSS历史峰值
    unsigned long hiwater_vm;       // VM历史峰值
    unsigned long rss;              // 实际驻留内存
    // ...
};
```

### vm_area_struct (VMA)

```c
struct vm_area_struct {
    unsigned long vm_start;         // 区域起始地址
    unsigned long vm_end;           // 区域结束地址

    // 链表和红黑树节点
    struct vm_area_struct *vm_next, *vm_prev;
    struct rb_node vm_rb;

    // 所属mm_struct
    struct mm_struct *vm_mm;

    // 访问权限
    unsigned long vm_flags;         // VM_READ, VM_WRITE, VM_EXEC...

    // 匿名映射或文件映射
    struct file *vm_file;           // 关联的文件
    unsigned long vm_pgoff;         // 文件偏移

    // 操作函数表
    const struct vm_operations_struct *vm_ops;

    // 私有数据
    void *vm_private_data;
    // ...
};
```

### vm_flags

```c
// VMA标志位
#define VM_NONE         0x00000000  // 无权限
#define VM_READ         0x00000001  // 可读
#define VM_WRITE        0x00000002  // 可写
#define VM_EXEC         0x00000004  // 可执行
#define VM_SHARED       0x00000008  // 共享映射

#define VM_MAYREAD      0x00000010  // 可能可读
#define VM_MAYWRITE     0x00000020  // 可能可写
#define VM_MAYEXEC      0x00000040  // 可能可执行
#define VM_MAYSHARE     0x00000080  // 可能共享

#define VM_GROWSDOWN    0x00000100  // 向下增长(栈)
#define VM_UFFD_MISSING 0x00000200  // Userfaultfd missing pages
#define VM_PFNMAP       0x00000400  // 页帧号映射
#define VM_UFFD_WP      0x00001000  // Userfaultfd write-protect

#define VM_LOCKED       0x00002000  // mlock锁定
#define VM_IO           0x00004000  // IO映射
#define VM_SEQ_READ     0x00008000  // 顺序读
#define VM_RAND_READ    0x00010000  // 随机读

#define VM_DONTCOPY     0x00020000  // fork时不复制
#define VM_DONTEXPAND   0x00040000  // 不可mremap扩展
#define VM_LOCKONFAULT  0x00080000  // mlock时按需锁定
#define VM_ACCOUNT      0x00100000  // 是否计入committed

#define VM_NORESERVE    0x00200000  // 不预留swap空间
#define VM_HUGETLB      0x00400000  // 大页映射
#define VM_ARCH_1       0x01000000  // 架构特定
#define VM_ARCH_2       0x02000000  // 架构特定

#define VM_DONTDUMP     0x04000000  // 不包含在core dump中
```

## VMA类型

| 类型 | 说明 | 示例 |
|------|------|------|
| 代码段 | 只读可执行 | .text段 |
| 数据段 | 读写 | .data, .bss |
| 堆 | 可读写，可增长 | malloc分配 |
| 栈 | 可读写，向下增长 | 局部变量 |
| 文件映射 | 与文件关联 | mmap文件 |
| 匿名映射 | 无文件关联 | malloc大块 |
| 共享内存 | 进程间共享 | shm |

## VMA查找与操作

### 查找VMA

```c
// 查找包含地址addr的VMA
struct vm_area_struct *find_vma(struct mm_struct *mm,
                                unsigned long addr);

// 查找起始地址>=addr的VMA
struct vm_area_struct *find_vma_prev(struct mm_struct *mm,
                                     unsigned long addr,
                                     struct vm_area_struct **pprev);

// 查找与区间[first,last)相交的VMA
struct vm_area_struct *find_vma_intersection(struct mm_struct *mm,
                                             unsigned long start,
                                             unsigned long end);
```

### 查找示例

```c
// 检查地址是否有效
unsigned long addr = 0x400000;
struct vm_area_struct *vma = find_vma(current->mm, addr);

if (vma && vma->vm_start <= addr) {
    // 地址在VMA范围内
    if (vma->vm_flags & VM_READ) {
        // 可读
    }
} else {
    // 地址无效
}
```

## 地址空间操作

### 获取进程地址空间

```c
// 获取当前进程的mm
struct mm_struct *mm = current->mm;

// 获取并增加引用计数
struct mm_struct *mm = get_task_mm(task);
mmput(mm);  // 释放引用
```

### 锁定地址空间

```c
// 读锁（允许多个读者）
mmap_read_lock(mm);
// 访问VMA...
mmap_read_unlock(mm);

// 写锁（独占）
mmap_write_lock(mm);
// 修改VMA...
mmap_write_unlock(mm);
```

## 进程地址空间查看

```bash
# 查看进程内存映射
cat /proc/<pid>/maps

# 查看进程内存统计
cat /proc/<pid>/status | grep -i vm

# 详细的内存映射信息
pmap -x <pid>
```
