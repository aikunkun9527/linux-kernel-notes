# 系统调用入口详解

> [!info] 目标
> 深入理解系统调用从用户态到内核态的完整入口流程

## 系统调用方式对比

### x86架构系统调用机制

```
系统调用方式演进：

8086/80286    int 0x80          传统方式，兼容性好
─────────────────────────────────────────────────
Pentium Pro   sysenter/sysexit  快速系统调用（32位）
─────────────────────────────────────────────────
AMD64         syscall/sysret    快速系统调用（64位）
─────────────────────────────────────────────────
```

### 性能对比

| 方式 | 入口开销 | 适用场景 | 特点 |
|------|----------|----------|------|
| int 0x80 | ~300周期 | 兼容模式 | 最慢，最兼容 |
| sysenter | ~100周期 | 32位原生 | 中等速度 |
| syscall | ~50周期 | 64位原生 | 最快，现代标准 |

## syscall指令（x86_64）

### CPU硬件行为

```
syscall指令执行时CPU自动完成：

1. 保存用户态状态
   - RCX ← RIP（返回地址）
   - R11 ← RFLAGS（标志寄存器）

2. 加载内核态状态
   - RIP ← IA32_LSTAR MSR
   - CS ← IA32_STAR MSR[47:32]
   - SS ← IA32_STAR MSR[47:32] + 8

3. 切换特权级
   - CPL ← 0（进入Ring 0）
   - RFLAGS ← RFLAGS & ~(TF|IF|DF|NT)

4. 切换栈（由软件完成）
   - 需要swapgs指令配合
```

### MSR寄存器配置

```c
// 相关MSR寄存器
#define MSR_IA32_STAR           0xC0000081  // CS/SS选择子
#define MSR_IA32_LSTAR          0xC0000082  // 64位syscall入口
#define MSR_IA32_CSTAR          0xC0000083  // 32位兼容模式入口
#define MSR_IA32_FMASK          0xC0000084  // RFLAGS掩码

// 内核初始化syscall MSR
void syscall_init(void)
{
    // 设置CS/SS选择子
    // STAR[47:32] = 内核CS
    // STAR[31:16] = 用户CS
    wrmsr(MSR_IA32_STAR, 0, (__USER32_CS << 16) | __KERNEL_CS);

    // 设置syscall入口地址
    wrmsrl(MSR_IA32_LSTAR, (unsigned long)entry_SYSCALL_64);

    // 设置RFLAGS掩码
    wrmsrl(MSR_IA32_FMASK, X86_EFLAGS_TF|X86_EFLAGS_DF|X86_EFLAGS_IF|X86_EFLAGS_NT);
}
```

## entry_SYSCALL_64详解

### 完整入口流程

```asm
// arch/x86/entry/entry_64.S
ENTRY(entry_SYSCALL_64)
    /*
     * 1. 交换GS基址
     * swapgs将当前GS基址与IA32_KERNEL_GS_BASE MSR交换
     * 用户态GS指向用户数据，内核态GS指向内核per-cpu数据
     */
    swapgs

    /*
     * 2. 保存用户态栈指针
     * 并切换到内核栈
     */
    movq    %rsp, PER_CPU_VAR(cpu_tss_rw + TSS_sp2)
    movq    PER_CPU_VAR(cpu_current_top_of_stack), %rsp

    /*
     * 3. 构建pt_regs结构
     * 在内核栈上保存用户态寄存器
     */
    /* 保存用户态RIP, CS, RFLAGS, RSP, SS */
    pushq   $__USER_DS              /* SS */
    pushq   PER_CPU_VAR(cpu_tss_rw + TSS_sp2)  /* RSP */
    pushq   %r11                    /* RFLAGS (保存在R11中) */
    pushq   $__USER_CS              /* CS */
    pushq   %rcx                    /* RIP (保存在RCX中) */

    /* 保存通用寄存器 */
    pushq   %rax                    /* 系统调用号 */
    pushq   %rdi                    /* 参数1 */
    pushq   %rsi                    /* 参数2 */
    pushq   %rdx                    /* 参数3 */
    pushq   %r10                    /* 参数4 */
    pushq   %r8                     /* 参数5 */
    pushq   %r9                     /* 参数6 */

    /*
     * 4. 清除寄存器低位（安全）
     */
    movq    PER_CPU_VAR(pcpu_hot + 8), %r11
    movq    $0, PER_CPU_VAR(pcpu_hot + 8)

    /*
     * 5. 调用系统调用处理函数
     */
    movq    %rsp, %rdi              /* pt_regs指针作为第一个参数 */
    call    do_syscall_64

    /*
     * 6. 返回用户态
     */
    movq    %rax, RAX(%rsp)         /* 保存返回值 */

    /* 检查是否需要信号处理等 */
    EXIT_TO_USER_MODE

    /* 恢复寄存器并返回 */
    popq    %r9
    popq    %r8
    popq    %r10
    popq    %rdx
    popq    %rsi
    popq    %rdi
    popq    %rax

    popq    %rcx                     /* 恢复RIP */
    popq    %r11                     /* 恢复RFLAGS */
    popq    %rsp                     /* 恢复RSP */

    swapgs                           /* 切换回用户GS */
    sysretq                          /* 返回用户态 */
END(entry_SYSCALL_64)
```

### pt_regs结构

```c
// arch/x86/include/asm/ptrace.h
struct pt_regs {
    /*
     * C ABI says these registers are callee-clobbered.
     * 他们不会在函数调用间被保存
     */
    unsigned long r15;
    unsigned long r14;
    unsigned long r13;
    unsigned long r12;
    unsigned long rbp;
    unsigned long rbx;

    /* 这些由syscall入口保存 */
    unsigned long r11;      /* RFLAGS */
    unsigned long r10;      /* 参数4 */
    unsigned long r9;       /* 参数6 */
    unsigned long r8;       /* 参数5 */
    unsigned long rax;      /* 系统调用号/返回值 */
    unsigned long rcx;      /* RIP */
    unsigned long rdx;      /* 参数3 */
    unsigned long rsi;      /* 参数2 */
    unsigned long rdi;      /* 参数1 */
    unsigned long orig_rax; /* 原始系统调用号 */

    /* 用户态上下文 */
    unsigned long rip;      /* 用户态指令指针 */
    unsigned long cs;       /* 用户态代码段 */
    unsigned long eflags;   /* 用户态标志 */
    unsigned long rsp;      /* 用户态栈指针 */
    unsigned long ss;       /* 用户态栈段 */
};
```

## do_syscall_64

```c
// arch/x86/entry/common.c
__visible noinstr void do_syscall_64(struct pt_regs *regs, int nr)
{
    // 添加到系统调用追踪
    add_random_kstack_offset();
    nr = syscall_enter_from_user_mode(regs, nr);

    // 检查系统调用号是否有效
    if (likely(nr < NR_syscalls)) {
        nr = array_index_nospec(nr, NR_syscalls);
        regs->ax = sys_call_table[nr](regs);  // 调用处理函数
    }

    // 系统调用退出处理
    syscall_exit_to_user_mode(regs);
}
```

## int 0x80入口（兼容模式）

### 中断门方式

```asm
// arch/x86/entry/entry_64_compat.S
ENTRY(entry_INT80_compat)
    /*
     * 32位兼容模式的系统调用
     */
    pushq   $__USER32_DS          /* SS */
    pushq   %rsp                  /* RSP (用户态) */
    pushq   %r11                  /* RFLAGS */
    pushq   $__USER32_CS          /* CS */
    pushq   %rcx                  /* RIP */

    /* 保存寄存器 */
    pushq   %rax
    pushq   %rdi
    pushq   %rsi
    pushq   %rdx
    pushq   %rcx                  /* 参数4在32位是ebp */
    pushq   %rbx                  /* 参数5 */
    pushq   %rbp                  /* 参数6 */

    /* 调用处理函数 */
    movq    %rsp, %rdi
    call    do_int80_syscall_32

    /* 返回 */
    jmp     swapgs_restore_regs_and_return_to_usermode
END(entry_INT80_compat)
```

### 参数传递差异

```c
// int 0x80方式（32位）
// 参数通过栈传递，由内核从用户栈读取

// eax = 系统调用号
// ebx = 参数1, ecx = 参数2, edx = 参数3
// esi = 参数4, edi = 参数5, ebp = 参数6
```

## swapgs指令

### 工作原理

```
swapgs行为：
┌─────────────────────────────────────────────────────┐
│ 用户态执行swapgs前：                                 │
│   GS.base → 用户数据（如TLS）                        │
│   IA32_KERNEL_GS_BASE MSR → 内核per-cpu数据          │
│                                                      │
│ swapgs执行后：                                       │
│   GS.base → 内核per-cpu数据                          │
│   IA32_KERNEL_GS_BASE MSR → 用户数据                 │
└─────────────────────────────────────────────────────┘
```

### per-cpu数据访问

```c
// 通过GS访问per-cpu变量
#define PER_CPU_VAR(var)    %gs:var

// 示例：获取当前进程
static __always_inline struct task_struct *get_current(void)
{
    return this_cpu_read_stable(current_task);
}

// 展开为
mov %gs:current_task, %rax
```

## 返回用户态

### sysret指令

```
sysret执行时：

1. RIP ← RCX（返回地址）
2. RFLAGS ← R11（恢复标志）
3. CPL ← 3（返回Ring 3）
4. CS/SS ← 从STAR MSR恢复
```

### iret指令（备用）

```asm
// 更通用的返回方式（可用于中断返回）
iretq

// 从栈上恢复：
// RIP, CS, RFLAGS, RSP, SS
```

## 系统调用入口图解

```
用户态                              内核态
  │                                   │
  │ mov rax, syscall_nr               │
  │ mov rdi, arg1                     │
  │ ...                               │
  │ syscall ─────────────────────────►│
  │                                   │
  │              ┌────────────────────┤ swapgs
  │              │                    │ 切换内核栈
  │              │                    │
  │              │                    │ push pt_regs
  │              │                    │
  │              │                    │ call do_syscall_64
  │              │                    │     │
  │              │                    │     ▼
  │              │                    │ sys_call_table[nr]
  │              │                    │     │
  │              │                    │     ▼
  │              │                    │ sys_xxx()
  │              │                    │     │
  │              │                    │ ◄───┘
  │              │                    │
  │◄──────────────────────────────────┤ sysret
  │                                   │
  ▼                                   ▼
用户程序继续执行                  返回内核
```
