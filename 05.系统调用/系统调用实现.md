## SYSCALL_DEFINE宏

### 宏的作用

```c
// 为什么使用SYSCALL_DEFINE而不是直接定义函数？
// 1. 安全性：防止参数泄露
// 2. 追踪：支持ftrace/audit
// 3. 兼容性：处理不同架构的差异
```

### 宏的定义

```c
// include/linux/syscalls.h

// 基础宏
#define SYSCALL_DEFINE1(name, ...) SYSCALL_DEFINEx(1, name, __VA_ARGS__)
#define SYSCALL_DEFINE2(name, ...) SYSCALL_DEFINEx(2, name, __VA_ARGS__)
#define SYSCALL_DEFINE3(name, ...) SYSCALL_DEFINEx(3, name, __VA_ARGS__)
#define SYSCALL_DEFINE4(name, ...) SYSCALL_DEFINEx(4, name, __VA_ARGS__)
#define SYSCALL_DEFINE5(name, ...) SYSCALL_DEFINEx(5, name, __VA_ARGS__)
#define SYSCALL_DEFINE6(name, ...) SYSCALL_DEFINEx(6, name, __VA_ARGS__)

// 展开宏
#define SYSCALL_DEFINEx(x, name, ...)                          \
    SYSCALL_METADATA(name, x, __VA_ARGS__)                     \
    __SYSCALL_DEFINEx(x, name, __VA_ARGS__)

#define __SYSCALL_DEFINEx(x, name, ...)                        \
    asmlinkage long sys##name(__MAP(x,__SC_LONG,__VA_ARGS__)); \
    asmlinkage long __se_sys##name(__MAP(x,__SC_LONG,__VA_ARGS__)); \
    static inline long __do_sys##name(__MAP(x,__SC_DECL,__VA_ARGS__));\
    asmlinkage long sys##name(__MAP(x,__SC_LONG,__VA_ARGS__))  \
    {                                                          \
        return __se_sys##name(__MAP(x,__SC_ARGS,__VA_ARGS__)); \
    }                                                          \
    asmlinkage long __se_sys##name(__MAP(x,__SC_LONG,__VA_ARGS__))\
    {                                                          \
        long ret = __do_sys##name(__MAP(x,__SC_CAST,__VA_ARGS__));\
        __MAP(x,__SC_TEST,__VA_ARGS__);                        \
        __PROTECT(x, ret,__MAP(x,__SC_ARGS,__VA_ARGS__));      \
        return ret;                                            \
    }                                                          \
    static inline long __do_sys##name(__MAP(x,__SC_DECL,__VA_ARGS__)
```

## 典型系统调用实现

### sys_read

```c
// fs/read_write.c
SYSCALL_DEFINE3(read, unsigned int, fd, char __user *, buf, size_t, count)
{
    return ksys_read(fd, buf, count);
}

ssize_t ksys_read(unsigned int fd, char __user *buf, size_t count)
{
    struct fd f = fdget_pos(fd);
    ssize_t ret = -EBADF;

    if (f.file) {
        loff_t pos, *ppos = file_ppos(f.file);
        if (ppos) {
            pos = *ppos;
            ppos = &pos;
        }
        ret = vfs_read(f.file, buf, count, ppos);
        if (ppos)
            f.file->f_pos = pos;
        fdput_pos(f);
    }
    return ret;
}

ssize_t vfs_read(struct file *file, char __user *buf, size_t count, loff_t *pos)
{
    ssize_t ret;

    // 检查权限
    if (!(file->f_mode & FMODE_READ))
        return -EBADF;
    if (!(file->f_mode & FMODE_CAN_READ))
        return -EINVAL;

    // 检查用户空间指针
    ret = rw_verify_area(READ, file, pos, count);
    if (ret)
        return ret;

    // 调用文件系统的read方法
    if (file->f_op->read)
        ret = file->f_op->read(file, buf, count, pos);
    else if (file->f_op->read_iter)
        ret = new_sync_read(file, buf, count, pos);
    else
        ret = -EINVAL;

    return ret;
}
```

### sys_openat

```c
// fs/open.c
SYSCALL_DEFINE4(openat, int, dfd, const char __user *, filename, int, flags, umode_t, mode)
{
    return do_sys_open(dfd, filename, flags, mode);
}

static long do_sys_open(int dfd, const char __user *filename, int flags, umode_t mode)
{
    struct open_flags op;
    int fd = build_open_flags(flags, mode, &op);
    struct filename *tmp;

    if (fd)
        return fd;

    // 复制文件名到内核
    tmp = getname(filename);
    if (IS_ERR(tmp))
        return PTR_ERR(tmp);

    // 分配文件描述符
    fd = get_unused_fd_flags(flags);
    if (fd >= 0) {
        struct file *f = do_filp_open(dfd, tmp, &op);
        if (IS_ERR(f)) {
            put_unused_fd(fd);
            fd = PTR_ERR(f);
        } else {
            fsnotify_open(f);
            fd_install(fd, f);  // 关联fd和file
        }
    }
    putname(tmp);
    return fd;
}
```

### sys_mmap

```c
// mm/mmap.c
SYSCALL_DEFINE6(mmap, unsigned long, addr, unsigned long, len,
                unsigned long, prot, unsigned long, flags,
                unsigned long, fd, unsigned long, off)
{
    return ksys_mmap_pgoff(addr, len, prot, flags, fd, off >> PAGE_SHIFT);
}

unsigned long ksys_mmap_pgoff(unsigned long addr, unsigned long len,
                              unsigned long prot, unsigned long flags,
                              unsigned long fd, unsigned long pgoff)
{
    struct file *file = NULL;
    unsigned long retval;

    // 检查参数
    if (!(flags & MAP_ANONYMOUS)) {
        // 文件映射
        file = fget(fd);
        if (!file)
            return -EBADF;
    }

    // 调用vm_mmap_pgoff
    retval = vm_mmap_pgoff(file, addr, len, prot, flags, pgoff);

    if (file)
        fput(file);

    return retval;
}
```

### sys_execve

```c
// fs/exec.c
SYSCALL_DEFINE3(execve, const char __user *, filename,
                const char __user *const __user *, argv,
                const char __user *const __user *, envp)
{
    return do_execve(getname(filename), argv, envp);
}

int do_execve(struct filename *filename,
              const char __user *const __user *__argv,
              const char __user *const __user *__envp)
{
    struct linux_binprm *bprm;
    int retval;

    // 分配bprm结构
    bprm = alloc_bprm();
    if (!bprm)
        return -ENOMEM;

    // 复制参数和环境变量
    bprm->argc = count(argv, MAX_ARG_STRINGS);
    bprm->envc = count(envp, MAX_ARG_STRINGS);

    // 复制到内核
    copy_strings(bprm->envc, envp, bprm);
    copy_strings(bprm->argc, argv, bprm);

    // 执行程序
    retval = exec_binprm(bprm);

    return retval;
}
```

## 用户空间数据访问

### copy_from_user / copy_to_user

```c
#include <linux/uaccess.h>

// 从用户空间复制到内核
unsigned long copy_from_user(void *to, const void __user *from, unsigned long n);

// 从内核复制到用户空间
unsigned long copy_to_user(void __user *to, const void *from, unsigned long n);

// 返回值：未复制的字节数（0表示全部成功）
```

### 安全检查函数

```c
// 检查用户空间地址是否有效
#define access_ok(addr, size) \
    __access_ok((unsigned long)(addr), (size))

// 获取/设置单个值
int get_user(x, const void __user *ptr);
int put_user(x, void __user *ptr);

// 安全字符串复制
long strncpy_from_user(char *dst, const char __user *src, long count);

// 获取用户空间字符串长度
long strnlen_user(const char __user *s, long n);
```

### 示例：安全的系统调用实现

```c
SYSCALL_DEFINE2(my_copy, const char __user *, src, char __user *, dst, size_t, len)
{
    char *kbuf;
    int ret = 0;

    // 1. 检查用户空间指针
    if (!access_ok(src, len) || !access_ok(dst, len))
        return -EFAULT;

    // 2. 分配内核缓冲区
    kbuf = kmalloc(len, GFP_KERNEL);
    if (!kbuf)
        return -ENOMEM;

    // 3. 从用户空间复制
    if (copy_from_user(kbuf, src, len)) {
        ret = -EFAULT;
        goto out;
    }

    // 4. 处理数据（可选）
    // ...

    // 5. 复制回用户空间
    if (copy_to_user(dst, kbuf, len)) {
        ret = -EFAULT;
        goto out;
    }

    ret = len;
out:
    kfree(kbuf);
    return ret;
}
```

## 系统调用返回值

### 返回值约定

```c
// 成功返回：非负值（通常0或正数）
// 错误返回：负的错误码

// 错误码定义（include/uapi/asm-generic/errno-base.h）
#define EPERM        1  /* Operation not permitted */
#define ENOENT       2  /* No such file or directory */
#define ESRCH        3  /* No such process */
#define EINTR        4  /* Interrupted system call */
#define EIO          5  /* I/O error */
#define ENXIO        6  /* No such device or address */
#define E2BIG        7  /* Argument list too long */
#define ENOEXEC      8  /* Exec format error */
#define EBADF        9  /* Bad file number */
#define ECHILD      10  /* No child processes */
#define EAGAIN      11  /* Try again */
#define ENOMEM      12  /* Out of memory */
#define EACCES      13  /* Permission denied */
#define EFAULT      14  /* Bad address */
#define EINVAL      22  /* Invalid argument */
```

### 返回值处理

```c
// 内核返回
SYSCALL_DEFINE1(my_syscall, int, arg)
{
    if (arg < 0)
        return -EINVAL;  // 返回负错误码

    return 0;  // 成功返回0
}

// glibc处理
// 错误时设置errno，返回-1
long result = syscall(__NR_my_syscall, arg);
if (result < 0) {
    errno = -result;  // 错误码取反
    return -1;
}
return result;
```
