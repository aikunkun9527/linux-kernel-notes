## 🧪 实验一：使用strace跟踪系统调用

> [!tip] 目标
> 学习使用strace工具分析程序的系统调用行为

### 1.1 基本使用

```bash
# 跟踪程序的所有系统调用
strace ls -la

# 输出示例
execve("/usr/bin/ls", ["ls", "-la"], 0x7fff...) = 0
brk(NULL)                               = 0x55a5a6b60000
access("/etc/ld.so.preload", R_OK)      = -1 ENOENT (No such file...)
openat(AT_FDCWD, "/etc/ld.so.cache", O_RDONLY|O_CLOEXEC) = 3
fstat(3, {st_mode=S_IFREG|0644, st_size=...}) = 0
mmap(NULL, 89507, PROT_READ, MAP_PRIVATE, 3, 0) = 0x7f...
close(3)                                = 0
...
write(1, "total 16\ndrwxr-xr-x  2 roo"..., 64) = 64
exit_group(0)                           = ?
```

### 1.2 过滤特定系统调用

```bash
# 只跟踪文件相关系统调用
strace -e trace=openat,read,write,close ./program

# 只跟踪网络相关
strace -e trace=network ./program

# 只跟踪内存相关
strace -e trace=mmap,munmap,brk ./program

# 排除某些调用
strace -e trace=!read,write ./program
```

### 1.3 统计系统调用

```bash
# 生成系统调用统计
strace -c ./program

# 输出示例
% time     seconds  usecs/call     calls    errors syscall
------ ----------- ----------- --------- --------- ----------------
 30.00    0.000300           3       100           write
 25.00    0.000250           2       125        10 openat
 20.00    0.000200           1       200           read
 15.00    0.000150           3        50           mmap
 10.00    0.000100           2        50           close
------ ----------- ----------- --------- --------- ----------------
100.00    0.001000                   525        10 total
```

### 1.4 跟踪运行中的进程

```bash
# 找到进程ID
ps aux | grep my_program

# 附加到运行中的进程
strace -p <PID>

# 跟踪特定线程
strace -p <TID>  # 使用线程ID
```

### 1.5 详细输出

```bash
# 显示字符串内容
strace -s 200 ./program

# 显示时间戳
strace -t ./program

# 显示相对时间
strace -r ./program

# 显示耗时
strace -T ./program
```

## 🧪 实验二：直接使用syscall

> [!tip] 目标
> 绕过glibc，直接使用syscall指令

### 2.1 C语言直接调用

```c
// 文件: exp2_direct_syscall.c
#define _GNU_SOURCE
#include <unistd.h>
#include <sys/syscall.h>
#include <sys/types.h>
#include <stdio.h>
#include <string.h>

int main() {
    const char *msg = "Hello, syscall!\n";

    // 方法1: 使用glibc的syscall包装
    // write(1, msg, strlen(msg));

    // 方法2: 直接使用syscall
    ssize_t ret = syscall(SYS_write, 1, msg, strlen(msg));

    printf("syscall返回值: %zd\n", ret);

    // 获取进程ID
    pid_t pid = syscall(SYS_getpid);
    printf("PID: %d\n", pid);

    return 0;
}
```

### 2.2 内联汇编调用

```c
// 文件: exp2_inline_syscall.c
#include <stdio.h>
#include <unistd.h>

// x86_64系统调用号
#define __NR_write  1
#define __NR_getpid 39

static inline long my_syscall1(long n, long a1) {
    long ret;
    __asm__ volatile (
        "syscall"
        : "=a"(ret)
        : "a"(n), "D"(a1)
        : "rcx", "r11", "memory"
    );
    return ret;
}

static inline long my_syscall3(long n, long a1, long a2, long a3) {
    long ret;
    __asm__ volatile (
        "syscall"
        : "=a"(ret)
        : "a"(n), "D"(a1), "S"(a2), "d"(a3)
        : "rcx", "r11", "memory"
    );
    return ret;
}

int main() {
    const char *msg = "Hello from inline syscall!\n";

    // 直接使用汇编syscall
    my_syscall3(__NR_write, 1, (long)msg, 20);

    pid_t pid = my_syscall1(__NR_getpid, 0);
    printf("PID from inline: %d\n", pid);

    return 0;
}
```

### 2.3 纯汇编系统调用

```asm
# 文件: exp2_asm_syscall.S
.section .data
msg:
    .ascii "Hello from assembly!\n"
    .set msg_len, . - msg

.section .text
.global _start

_start:
    # write(1, msg, msg_len)
    mov $1, %rax           # syscall number for write
    mov $1, %rdi           # fd = stdout
    lea msg(%rip), %rsi    # buf
    mov $msg_len, %rdx     # count
    syscall

    # exit(0)
    mov $60, %rax          # syscall number for exit
    xor %rdi, %rdi         # status = 0
    syscall
```

```bash
# 编译和运行
as -o exp2_asm_syscall.o exp2_asm_syscall.S
ld -o exp2_asm_syscall exp2_asm_syscall.o
./exp2_asm_syscall
```

## 🧪 实验三：添加新系统调用

> [!tip] 目标
> 在内核中添加一个自定义系统调用

### 3.1 创建内核模块方式

```c
// 文件: exp3_new_syscall/mysyscall.c
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/init.h>
#include <linux/syscalls.h>
#include <linux/uaccess.h>

// 系统调用实现
asmlinkage long sys_mysyscall(int value, char __user *buf, size_t len)
{
    char kbuf[64];
    int ret;

    pr_info("mysyscall called: value=%d\n", value);

    // 检查用户空间指针
    if (buf && len > 0) {
        if (len > sizeof(kbuf) - 1)
            len = sizeof(kbuf) - 1;

        // 从用户空间复制
        if (copy_from_user(kbuf, buf, len))
            return -EFAULT;

        kbuf[len] = '\0';
        pr_info("received: %s\n", kbuf);

        // 返回数据到用户空间
        snprintf(kbuf, sizeof(kbuf), "Kernel: received %d", value);
        if (copy_to_user(buf, kbuf, strlen(kbuf) + 1))
            return -EFAULT;
    }

    return value * 2;  // 返回值
}

// 注意：现代内核不推荐动态添加系统调用
// 这里仅作学习演示

MODULE_LICENSE("GPL");
MODULE_AUTHOR("Student");
MODULE_DESCRIPTION("系统调用学习模块");
```

### 3.2 修改内核源码方式

#### 步骤1：实现系统调用

```c
// 文件: kernel/mysyscall.c
#include <linux/kernel.h>
#include <linux/syscalls.h>
#include <linux/uaccess.h>

SYSCALL_DEFINE3(mysyscall, int, value, char __user *, buf, size_t, len)
{
    pr_info("mysyscall: value=%d, len=%zu\n", value, len);

    if (value < 0)
        return -EINVAL;

    return value * 2;
}
```

#### 步骤2：添加到系统调用表

```c
// 文件: arch/x86/entry/syscalls/syscall_64.tbl
// 添加一行（使用合适的编号）
451    common    mysyscall    sys_mysyscall
```

#### 步骤3：添加函数声明

```c
// 文件: include/linux/syscalls.h
asmlinkage long sys_mysyscall(int value, char __user *buf, size_t len);
```

#### 步骤4：更新Makefile

```makefile
# 文件: kernel/Makefile
# 添加 mysyscall.o
obj-y += mysyscall.o
```

#### 步骤5：编译内核

```bash
make -j$(nproc)
make modules_install
make install
reboot
```

## 🧪 实验四：系统调用性能测试

> [!tip] 目标
> 测量系统调用的开销

### 4.1 系统调用开销测量

```c
// 文件: exp4_syscall_perf.c
#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <time.h>
#include <sys/syscall.h>

#define ITERATIONS 1000000

// 获取纳秒级时间
static inline long long get_ns() {
    struct timespec ts;
    clock_gettime(CLOCK_MONOTONIC, &ts);
    return ts.tv_sec * 1000000000LL + ts.tv_nsec;
}

int main() {
    long long start, end;
    long long total = 0;

    printf("系统调用性能测试 (%d次)\n", ITERATIONS);

    // 测试getpid系统调用
    start = get_ns();
    for (int i = 0; i < ITERATIONS; i++) {
        syscall(SYS_getpid);
    }
    end = get_ns();

    printf("getpid平均耗时: %.2f ns\n",
           (double)(end - start) / ITERATIONS);

    // 测试write系统调用
    start = get_ns();
    for (int i = 0; i < ITERATIONS; i++) {
        syscall(SYS_write, -1, NULL, 0);  // 写无效fd
    }
    end = get_ns();

    printf("write平均耗时: %.2f ns\n",
           (double)(end - start) / ITERATIONS);

    // 测试空函数调用（对比）
    volatile int x = 0;
    start = get_ns();
    for (int i = 0; i < ITERATIONS; i++) {
        x = x + 1;
    }
    end = get_ns();

    printf("空循环平均耗时: %.2f ns\n",
           (double)(end - start) / ITERATIONS);

    return 0;
}
```

### 4.2 不同系统调用方式对比

```c
// 文件: exp4_syscall_methods.c
#define _GNU_SOURCE
#include <stdio.h>
#include <unistd.h>
#include <sys/syscall.h>
#include <time.h>

#define ITERATIONS 100000

static inline long long get_ns() {
    struct timespec ts;
    clock_gettime(CLOCK_MONOTONIC, &ts);
    return ts.tv_sec * 1000000000LL + ts.tv_nsec;
}

int main() {
    long long start, end;
    pid_t pid;

    // 方法1: glibc包装
    start = get_ns();
    for (int i = 0; i < ITERATIONS; i++) {
        pid = getpid();
    }
    end = get_ns();
    printf("glibc getpid: %.2f ns\n", (double)(end - start) / ITERATIONS);

    // 方法2: syscall函数
    start = get_ns();
    for (int i = 0; i < ITERATIONS; i++) {
        pid = syscall(SYS_getpid);
    }
    end = get_ns();
    printf("syscall() getpid: %.2f ns\n", (double)(end - start) / ITERATIONS);

    // 方法3: 内联汇编
    start = get_ns();
    for (int i = 0; i < ITERATIONS; i++) {
        __asm__ volatile (
            "syscall"
            : "=a"(pid)
            : "a"(39)  // __NR_getpid
            : "rcx", "r11", "memory"
        );
    }
    end = get_ns();
    printf("inline asm getpid: %.2f ns\n", (double)(end - start) / ITERATIONS);

    return 0;
}
```

## 🧪 实验五：系统调用追踪与分析

> [!tip] 目标
> 使用内核追踪工具分析系统调用

### 5.1 使用ftrace

```bash
# 启用系统调用追踪
echo 1 > /sys/kernel/debug/tracing/events/syscalls/enable

# 只追踪特定系统调用
echo 1 > /sys/kernel/debug/tracing/events/syscalls/sys_enter_write/enable
echo 1 > /sys/kernel/debug/tracing/events/syscalls/sys_exit_write/enable

# 查看追踪结果
cat /sys/kernel/debug/tracing/trace_pipe

# 运行测试程序
./your_program &

# 禁用追踪
echo 0 > /sys/kernel/debug/tracing/events/syscalls/enable
```

### 5.2 使用eBPF/bpftrace

```bash
# 安装bpftrace
sudo apt install bpftrace

# 追踪所有系统调用
sudo bpftrace -e 'tracepoint:syscalls:sys_enter_* { @[args->id] = count(); }'

# 追踪特定进程的系统调用
sudo bpftrace -e 'tracepoint:syscalls:sys_enter_* /pid == 1234/ { printf("%s\n", probe); }'

# 统计系统调用耗时
sudo bpftrace -e '
tracepoint:syscalls:sys_enter_read { @start[tid] = nsecs; }
tracepoint:syscalls:sys_exit_read /@start[tid]/ {
    @read_ns = hist(nsecs - @start[tid]);
    delete(@start[tid]);
}
'
```

### 5.3 使用perf

```bash
# 记录系统调用
sudo perf record -e 'syscalls:sys_enter_*' ./program

# 查看报告
sudo perf report

# 实时统计
sudo perf stat -e 'syscalls:sys_enter_read,syscalls:sys_enter_write' ./program
```
