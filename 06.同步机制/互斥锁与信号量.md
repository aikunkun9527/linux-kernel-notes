# 互斥锁与信号量

> [!info] 目标
> 理解Linux内核互斥锁和信号量的实现与使用

## 概述

互斥锁和信号量是**可睡眠**的同步机制：
- 获取失败时，进程**睡眠等待**而非忙等
- 适用于**长时间持有**或**可能睡眠**的临界区
- 只能在**进程上下文**中使用

## 互斥锁 vs 信号量

| 特性 | 互斥锁 (mutex) | 信号量 (semaphore) |
|------|---------------|-------------------|
| 计数 | 二值 (0/1) | 可大于1 |
| 所有者 | 有所有者概念 | 无所有者 |
| 递归 | 不支持递归 | 不适用 |
| 用途 | 互斥访问 | 资源计数/同步 |
| 开销 | 较低 | 较高 |
| 优先 | 推荐使用 | 特定场景 |

## 互斥锁 (Mutex)

### 数据结构

```c
// include/linux/mutex.h
struct mutex {
    atomic_long_t        owner;      // 锁持有者
    spinlock_t           wait_lock;  // 保护等待队列
    struct list_head     wait_list;  // 等待队列
#ifdef CONFIG_DEBUG_MUTEXES
    void                *magic;
    struct task_struct  *owner_task;
    const char          *name;
    void                *dep_map;
#endif
};
```

### 初始化

```c
// 静态初始化
DEFINE_MUTEX(my_mutex);

// 动态初始化
struct mutex my_mutex;
mutex_init(&my_mutex);
```

### 基本操作

```c
#include <linux/mutex.h>

// 获取锁（可能睡眠）
void mutex_lock(struct mutex *lock);

// 释放锁
void mutex_unlock(struct mutex *lock);

// 尝试获取（非阻塞）
int mutex_trylock(struct mutex *lock);
// 返回: 1成功, 0失败

// 可中断获取
int mutex_lock_interruptible(struct mutex *lock);
// 返回: 0成功, -EINTR被信号中断

// 带超时获取
int mutex_lock_killable(struct mutex *lock);
```

### 使用示例

```c
static DEFINE_MUTEX(device_mutex);

int device_open(struct inode *inode, struct file *file)
{
    if (mutex_lock_interruptible(&device_mutex))
        return -ERESTARTSYS;

    // 临界区：设备初始化
    if (device_busy) {
        mutex_unlock(&device_mutex);
        return -EBUSY;
    }
    device_busy = 1;

    mutex_unlock(&device_mutex);
    return 0;
}

ssize_t device_read(struct file *file, char __user *buf,
                    size_t count, loff_t *ppos)
{
    if (mutex_lock_interruptible(&device_mutex))
        return -ERESTARTSYS;

    // 临界区：读取设备
    // 可以睡眠（如等待I/O）
    count = do_device_read(buf, count);

    mutex_unlock(&device_mutex);
    return count;
}
```

### 互斥锁内部实现

```
mutex_lock() 流程：

                    ┌─────────────┐
                    │ mutex_lock  │
                    └──────┬──────┘
                           │
                           ▼
               ┌─────────────────────┐
               │ atomic_cmpxchg      │
               │ 尝试无竞争获取        │
               └──────────┬──────────┘
                          │
              ┌───────────┴───────────┐
              │成功                   │失败
              ▼                       ▼
         返回成功           ┌──────────────────┐
                           │ __mutex_lock_slow│
                           └────────┬─────────┘
                                    │
                                    ▼
                           ┌──────────────────┐
                           │ 加入等待队列      │
                           │ 设置任务状态      │
                           │ schedule() 睡眠  │
                           └────────┬─────────┘
                                    │
                                    │ 被唤醒
                                    ▼
                           ┌──────────────────┐
                           │ 从等待队列移除     │
                           │ 获取锁成功        │
                           └──────────────────┘
```

## 信号量 (Semaphore)

### 数据结构

```c
// include/linux/semaphore.h
struct semaphore {
    raw_spinlock_t    lock;      // 保护信号量
    unsigned int      count;     // 计数值
    struct list_head  wait_list; // 等待队列
};
```

### 初始化

```c
// 静态初始化
DEFINE_SEMAPHORE(my_sem, 1);  // 初始值1（二值信号量）
DEFINE_SEMAPHORE(my_sem, 5);  // 初始值5（计数信号量）

// 动态初始化
struct semaphore my_sem;
sema_init(&my_sem, 1);
```

### 基本操作

```c
#include <linux/semaphore.h>

// 获取（P操作，可能睡眠）
void down(struct semaphore *sem);

// 可中断获取
int down_interruptible(struct semaphore *sem);
// 返回: 0成功, -EINTR被中断

// 带超时获取
int down_timeout(struct semaphore *sem, long timeout);
// 返回: 0成功, -ETIME超时

// 非阻塞尝试
int down_trylock(struct semaphore *sem);
// 返回: 0成功, 1失败

// 释放（V操作）
void up(struct semaphore *sem);
```

### 使用场景

#### 1. 资源计数

```c
// 有限资源管理（如5个连接池）
DEFINE_SEMAPHORE(connection_pool, 5);

int get_connection(void)
{
    if (down_interruptible(&connection_pool))
        return -ERESTARTSYS;

    // 获取连接
    return allocate_connection();
}

void release_connection(int conn)
{
    free_connection(conn);
    up(&connection_pool);  // 释放资源
}
```

#### 2. 同步（二值信号量）

```c
// 任务完成通知
DECLARE_COMPLETION(data_ready);  // 推荐用completion

// 或用信号量
static DEFINE_SEMAPHORE(data_ready, 0);

// 生产者
void produce_data(void)
{
    // 生产数据
    up(&data_ready);  // 通知消费者
}

// 消费者
void consume_data(void)
{
    down(&data_ready);  // 等待数据
    // 消费数据
}
```

## 读写信号量 (rw_semaphore)

### 特点

- 允许多个读者同时访问
- 写者独占访问
- 读者与写者互斥

### API

```c
#include <linux/rwsem.h>

// 初始化
DECLARE_RWSEM(my_rwsem);
// 或
struct rw_semaphore my_rwsem;
init_rwsem(&my_rwsem);

// 读锁
void down_read(struct rw_semaphore *sem);
int down_read_interruptible(struct rw_semaphore *sem);
int down_read_trylock(struct rw_semaphore *sem);
void up_read(struct rw_semaphore *sem);

// 写锁
void down_write(struct rw_semaphore *sem);
int down_write_interruptible(struct rw_semaphore *sem);
int down_write_trylock(struct rw_semaphore *sem);
void up_write(struct rw_semaphore *sem);

// 降级：写锁→读锁
void downgrade_write(struct rw_semaphore *sem);
```

### 使用示例

```c
static DECLARE_RWSEM(config_rwsem);

// 读取配置（多个读者可同时进行）
int get_config_value(const char *key)
{
    int value;

    down_read(&config_rwsem);
    value = lookup_config(key);
    up_read(&config_rwsem);

    return value;
}

// 更新配置（写者独占）
void update_config(const char *key, int value)
{
    down_write(&config_rwsem);
    set_config(key, value);
    up_write(&config_rwsem);
}

// 写后读场景
void update_and_read(const char *key, int value)
{
    down_write(&config_rwsem);
    set_config(key, value);

    // 降级为读锁，允许其他读者
    downgrade_write(&config_rwsem);

    // 此时持有读锁
    verify_config(key);

    up_read(&config_rwsem);
}
```

## 完成量 (Completion)

### 特点

- 用于**一次性的同步**
- 比**信号量更轻量**
- 任务完成通知场景

### API

```c
#include <linux/completion.h>

// 初始化
DECLARE_COMPLETION(my_comp);
// 或
struct completion my_comp;
init_completion(&my_comp);

// 等待完成
void wait_for_completion(struct completion *x);
int wait_for_completion_interruptible(struct completion *x);
int wait_for_completion_timeout(struct completion *x, unsigned long timeout);

// 发送完成信号
void complete(struct completion *x);      // 唤醒一个等待者
void complete_all(struct completion *x);  // 唤醒所有等待者

// 重置（可重用）
void reinit_completion(struct completion *x);
```

### 使用示例

```c
// 内核线程初始化完成通知
static DECLARE_COMPLETION(thread_started);

static int worker_thread(void *data)
{
    // 初始化
    setup_thread();

    // 通知主线程：已启动
    complete(&thread_started);

    // 工作循环
    while (!kthread_should_stop()) {
        // ...
    }
    return 0;
}

int start_worker(void)
{
    struct task_struct *task;

    task = kthread_run(worker_thread, NULL, "worker");
    if (IS_ERR(task))
        return PTR_ERR(task);

    // 等待线程启动完成
    wait_for_completion(&thread_started);

    return 0;
}
```

## 选择指南

```
需要保护临界区？
    │
    ├─ 临界区会睡眠？
    │     │
    │     ├─ 是 → 互斥锁或信号量
    │     │
    │     └─ 否 → 自旋锁
    │
    └─ 同步需求（非互斥）？
          │
          ├─ 任务完成通知 → completion
          │
          ├─ 资源计数 → semaphore
          │
          └─ 读写分离 → rw_semaphore
```

## 注意事项

> [!warning] 睡眠锁使用规则
> 1. **只能用于进程上下文**：中断上下文不能睡眠
> 2. **不要长时间持有**：影响系统响应
> 3. **注意锁顺序**：避免死锁
> 4. **检查返回值**：interruptible版本可能被信号中断

### 死锁示例

```c
// ❌ 死锁：锁顺序不一致
// 线程A
mutex_lock(&lock1);
mutex_lock(&lock2);  // 等待lock2

// 线程B
mutex_lock(&lock2);
mutex_lock(&lock1);  // 等待lock1 → 死锁！

// ✓ 正确：统一锁顺序
// 所有线程都按lock1→lock2的顺序获取
```
