# 原子操作

> [!info] 目标
> 理解Linux内核原子操作的原理与使用

## 什么是原子操作

**原子操作**：不可分割的操作，要么全部完成，要么完全不执行

### 为什么需要原子操作

```
非原子操作的问题（count++）：

步骤    CPU 0          CPU 1          内存中的值
────────────────────────────────────────────────
 1     read count                              0
 2                    read count               0
 3     increment (1)                           0
 4                    increment (1)            0
 5     write count                             1
 6                    write count              1

期望: count = 2
实际: count = 1  ← 数据丢失！

原子操作保证：

步骤    CPU 0          CPU 1          内存中的值
────────────────────────────────────────────────
 1     atomic_inc()  (等待)                  0
 2     (完成)                           →     1
 3                    atomic_inc()           1
 4                    (完成)           →     2

正确结果: count = 2 ✓
```

## 原子变量类型

### atomic_t

```c
// include/linux/types.h
typedef struct {
    int counter;
} atomic_t;

// 64位版本
typedef struct {
    long counter;
} atomic64_t;

// 64位系统上的long类型
typedef atomic64_t atomic_long_t;
```

### 初始化

```c
// 静态初始化
atomic_t v = ATOMIC_INIT(0);
atomic64_t v64 = ATOMIC64_INIT(0);

// 动态初始化
atomic_t v;
atomic_set(&v, 0);
```

## 原子操作API

### 基本操作

```c
#include <linux/atomic.h>

atomic_t v = ATOMIC_INIT(0);

// 读取
int val = atomic_read(&v);

// 设置
atomic_set(&v, 10);

// 加法
atomic_add(5, &v);    // v += 5

// 减法
atomic_sub(3, &v);    // v -= 3

// 自增
atomic_inc(&v);       // v++

// 自减
atomic_dec(&v);       // v--
```

### 带返回值的操作

```c
// 加并返回新值
int new_val = atomic_add_return(5, &v);

// 减并返回新值
int new_val = atomic_sub_return(3, &v);

// 自增并返回新值
int new_val = atomic_inc_return(&v);

// 自减并返回新值
int new_val = atomic_dec_return(&v);
```

### 条件操作

```c
// 条件加（结果<0则不加）
int result = atomic_add_unless(&v, 5, 0);  // 如果v!=0，则加5

// 自减并测试（减后是否为0）
int is_zero = atomic_dec_and_test(&v);

// 自减返回是否为负
int is_negative = atomic_sub_and_test(5, &v);

// 自增并测试是否为负
int is_negative = atomic_inc_and_test(&v);
```

### 交换操作

```c
// 交换（返回旧值）
int old_val = atomic_xchg(&v, new_val);

// 比较并交换（CAS）
// 如果v == old，则v = new，返回旧值
int actual = atomic_cmpxchg(&v, old_val, new_val);
if (actual == old_val) {
    // 交换成功
} else {
    // 失败，v已被其他线程修改
}
```

### atomic64_t 操作

```c
atomic64_t v64 = ATOMIC64_INIT(0);

atomic64_set(&v64, 1000);
long val = atomic64_read(&v64);
atomic64_add(100, &v64);
atomic64_inc(&v64);
long old = atomic64_xchg(&v64, 2000);
```

## 原子位操作

### 位操作API

```c
#include <asm/bitops.h>

unsigned long flags = 0;

// 设置位
set_bit(0, &flags);   // flags |= (1 << 0)

// 清除位
clear_bit(0, &flags); // flags &= ~(1 << 0)

// 翻转位
change_bit(0, &flags); // flags ^= (1 << 0)

// 测试位
int is_set = test_bit(0, &flags);

// 测试并设置（返回原值）
int was_set = test_and_set_bit(0, &flags);

// 测试并清除
int was_set = test_and_clear_bit(0, &flags);

// 测试并翻转
int was_set = test_and_change_bit(0, &flags);
```

### 位操作示例

```c
// 状态标志位
#define FLAG_INITIALIZED  0
#define FLAG_RUNNING      1
#define FLAG_ERROR        2
#define FLAG_STOPPED      3

unsigned long device_flags;

// 设置运行状态
set_bit(FLAG_RUNNING, &device_flags);

// 检查是否运行
if (test_bit(FLAG_RUNNING, &device_flags)) {
    // 正在运行
}

// 清除运行状态
clear_bit(FLAG_RUNNING, &device_flags);

// 设置错误并检查之前是否已错误
if (!test_and_set_bit(FLAG_ERROR, &device_flags)) {
    // 第一次设置错误，进行错误处理
}
```

## 原子操作实现原理

### x86实现

```c
// x86使用LOCK前缀实现原子性

// atomic_add实现
static inline void atomic_add(int i, atomic_t *v)
{
    asm volatile(LOCK_PREFIX "addl %1,%0"
                 : "+m" (v->counter)
                 : "ir" (i));
}

// LOCK_PREFIX定义
#ifdef CONFIG_SMP
#define LOCK_PREFIX "lock "
#else
#define LOCK_PREFIX ""
#endif

// lock前缀确保总线锁定，原子执行
```

### CMPXCHG指令

```c
// 比较并交换的实现
static inline int atomic_cmpxchg(atomic_t *v, int old, int new)
{
    int prev;

    asm volatile("lock; cmpxchgl %2,%1"
                 : "=a" (prev), "+m" (v->counter)
                 : "r" (new), "0" (old)
                 : "memory");

    return prev;
}

// cmpxchg指令：
// 比较 AL/AX/EAX/RAX 与 目标操作数
// 相等：ZF=1，目标 = 源操作数
// 不等：ZF=0，AL/AX/EAX/RAX = 目标操作数
```

### 内存屏障

```c
// 原子操作隐含内存屏障
// 确保操作顺序可见

// 显式屏障
smp_mb();    // 完整内存屏障
smp_rmb();   // 读屏障
smp_wmb();   // 写屏障
smp_mb__before_atomic();
smp_mb__after_atomic();
```

## 使用场景

### 1. 引用计数

```c
struct my_object {
    atomic_t refcount;
    // ...
};

struct my_object *obj_create(void)
{
    struct my_object *obj = kmalloc(sizeof(*obj), GFP_KERNEL);
    if (obj)
        atomic_set(&obj->refcount, 1);
    return obj;
}

void obj_get(struct my_object *obj)
{
    atomic_inc(&obj->refcount);
}

void obj_put(struct my_object *obj)
{
    if (atomic_dec_and_test(&obj->refcount)) {
        // 引用计数为0，释放对象
        kfree(obj);
    }
}
```

### 2. 状态标志

```c
struct device {
    atomic_t state;
    #define STATE_IDLE      0
    #define STATE_RUNNING   1
    #define STATE_STOPPING  2
};

void device_start(struct device *dev)
{
    if (atomic_cmpxchg(&dev->state, STATE_IDLE, STATE_RUNNING) == STATE_IDLE) {
        // 成功启动
        do_start(dev);
    }
}

void device_stop(struct device *dev)
{
    if (atomic_cmpxchg(&dev->state, STATE_RUNNING, STATE_STOPPING) == STATE_RUNNING) {
        do_stop(dev);
        atomic_set(&dev->state, STATE_IDLE);
    }
}
```

### 3. 无锁计数

```c
// 统计计数器
atomic_t event_count;

void event_happened(void)
{
    atomic_inc(&event_count);
}

int get_event_count(void)
{
    return atomic_read(&event_count);
}
```

### 4. 自旋锁基础

```c
// 简单自旋锁的实现基础
typedef struct {
    atomic_t locked;
} simple_spinlock_t;

void simple_lock(simple_spinlock_t *lock)
{
    while (atomic_xchg(&lock->locked, 1) == 1) {
        // 自旋等待
        cpu_relax();
    }
}

void simple_unlock(simple_spinlock_t *lock)
{
    atomic_set(&lock->locked, 0);
}
```

## atomic_t vs 普通变量

| 特性 | atomic_t | 普通变量 |
|------|----------|----------|
| 线程安全 | ✓ | ✗ |
| 性能开销 | 略高 | 最低 |
| 适用场景 | 并发访问 | 单线程或已加锁 |
| 代码复杂度 | 简单 | 需额外同步 |

## 常见陷阱

### 1. 非原子读取后操作

```c
// ❌ 错误：读取和操作之间可能被打断
int val = atomic_read(&v);
if (val > 10) {
    atomic_set(&v, val + 1);
}

// ✓ 正确：使用原子操作
if (atomic_read(&v) > 10) {
    atomic_inc(&v);  // 可能不符合原意，需要更精确的设计
}

// ✓ 正确：使用cmpxchg保证原子性
int old, new;
do {
    old = atomic_read(&v);
    if (old <= 10)
        break;
    new = old + 1;
} while (atomic_cmpxchg(&v, old, new) != old);
```

### 2. 多变量原子性

```c
// ❌ 错误：无法保证多变量原子性
atomic_t a, b;
atomic_set(&a, 1);
atomic_set(&b, 2);  // 这两步之间可能被打断

// ✓ 正确：使用锁保护多个变量
spinlock_t lock;
spin_lock(&lock);
a = 1;
b = 2;
spin_unlock(&lock);
```
