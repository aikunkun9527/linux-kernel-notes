## 🧪 实验一：原子操作与竞态条件

> [!tip] 目标
> 观察竞态条件，验证原子操作的必要性

### 1.1 竞态条件演示

```c
// 文件: exp1_race_condition.c
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/kthread.h>
#include <linux/delay.h>

static int shared_counter = 0;
static struct task_struct *threads[4];

static int increment_thread(void *data)
{
    int i;
    for (i = 0; i < 100000; i++) {
        shared_counter++;  // 非原子操作！
    }
    return 0;
}

static int __init race_init(void)
{
    int i;

    pr_info("Starting race condition test\n");

    // 创建4个线程同时增加计数器
    for (i = 0; i < 4; i++) {
        threads[i] = kthread_run(increment_thread, NULL, "inc_%d", i);
    }

    // 等待线程完成
    msleep(1000);

    // 期望值: 400000, 实际值会小于期望
    pr_info("Expected: 400000, Actual: %d\n", shared_counter);

    return 0;
}

static void __exit race_exit(void)
{
    pr_info("Module unloaded\n");
}

module_init(race_init);
module_exit(race_exit);
MODULE_LICENSE("GPL");
```

### 1.3 cmpxchg实现无锁算法

```c
// 文件: exp1_lockfree.c
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/atomic.h>

// 无锁栈实现
struct stack_node {
    int value;
    struct stack_node *next;
};

static struct stack_node *top = NULL;

static void push(int value)
{
    struct stack_node *new_node = kmalloc(sizeof(*new_node), GFP_KERNEL);
    struct stack_node *old_top;

    new_node->value = value;

    do {
        old_top = READ_ONCE(top);
        new_node->next = old_top;
    } while (cmpxchg(&top, old_top, new_node) != old_top);
}

static int pop(int *value)
{
    struct stack_node *old_top, *new_top;

    do {
        old_top = READ_ONCE(top);
        if (!old_top)
            return -1;  // 栈空

        new_top = old_top->next;
    } while (cmpxchg(&top, old_top, new_top) != old_top);

    *value = old_top->value;
    kfree(old_top);
    return 0;
}
```

## 🧪 实验二：自旋锁性能测试

> [!tip] 目标
> 测量自旋锁的性能开销

### 2.1 自旋锁性能测试模块

```c
// 文件: exp2_spinlock_perf.c
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/kthread.h>
#include <linux/spinlock.h>
#include <linux/time.h>
#include <linux/smp.h>

static DEFINE_SPINLOCK(test_lock);
static atomic_t test_counter = ATOMIC_INIT(0);

static ktime_t start_time, end_time;
static int test_iterations = 10000000;

static int lock_test_thread(void *data)
{
    int cpu = smp_processor_id();
    int i;

    pr_info("Thread starting on CPU %d\n", cpu);

    for (i = 0; i < test_iterations; i++) {
        spin_lock(&test_lock);
        // 极短临界区
        atomic_inc(&test_counter);
        spin_unlock(&test_lock);
    }

    return 0;
}

static int __init perf_init(void)
{
    int i, num_threads = 4;
    struct task_struct *threads[4];

    pr_info("Starting spinlock performance test\n");
    pr_info("Iterations per thread: %d\n", test_iterations);

    atomic_set(&test_counter, 0);
    start_time = ktime_get();

    for (i = 0; i < num_threads; i++) {
        threads[i] = kthread_create(lock_test_thread, NULL, "lock_test_%d", i);
        kthread_bind(threads[i], i);
        wake_up_process(threads[i]);
    }

    // 等待完成
    while (atomic_read(&test_counter) < test_iterations * num_threads)
        msleep(10);

    end_time = ktime_get();

    pr_info("Total time: %lld ns\n", ktime_to_ns(ktime_sub(end_time, start_time)));
    pr_info("Operations per second: %lld\n",
            (s64)num_threads * test_iterations * 1000000000LL /
            ktime_to_ns(ktime_sub(end_time, start_time)));

    return 0;
}

static void __exit perf_exit(void)
{
    pr_info("Module unloaded\n");
}

module_init(perf_init);
module_exit(perf_exit);
MODULE_LICENSE("GPL");
```

## 🧪 实验三：互斥锁与信号量对比

> [!tip] 目标
> 理解互斥锁和信号量的区别与使用场景

### 3.1 互斥锁示例

```c
// 文件: exp3_mutex.c
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/mutex.h>
#include <linux/kthread.h>
#include <linux/delay.h>

static DEFINE_MUTEX(resource_mutex);
static int shared_resource = 0;

static int producer_thread(void *data)
{
    int i;
    for (i = 0; i < 10; i++) {
        if (mutex_lock_interruptible(&resource_mutex))
            return -ERESTARTSYS;

        shared_resource++;
        pr_info("Producer: value = %d\n", shared_resource);

        mutex_unlock(&resource_mutex);
        msleep(100);
    }
    return 0;
}

static int consumer_thread(void *data)
{
    int i;
    for (i = 0; i < 10; i++) {
        if (mutex_lock_interruptible(&resource_mutex))
            return -ERESTARTSYS;

        pr_info("Consumer: value = %d\n", shared_resource);
        shared_resource--;

        mutex_unlock(&resource_mutex);
        msleep(150);
    }
    return 0;
}
```

### 3.2 信号量示例（资源池）

```c
// 文件: exp3_semaphore.c
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/semaphore.h>
#include <linux/kthread.h>

#define POOL_SIZE 3

static DEFINE_SEMAPHORE(resource_pool, POOL_SIZE);
static int resources_in_use = 0;

static int worker_thread(void *data)
{
    int id = *(int *)data;

    pr_info("Worker %d: waiting for resource\n", id);

    if (down_interruptible(&resource_pool)) {
        pr_info("Worker %d: interrupted\n", id);
        return -ERESTARTSYS;
    }

    resources_in_use++;
    pr_info("Worker %d: got resource (in use: %d/%d)\n", id, resources_in_use, POOL_SIZE);

    // 使用资源
    msleep(1000);

    resources_in_use--;
    pr_info("Worker %d: released resource\n", id);

    up(&resource_pool);
    return 0;
}

static int __init sem_init(void)
{
    struct task_struct *threads[5];
    int ids[5] = {1, 2, 3, 4, 5};
    int i;

    pr_info("Starting semaphore test (pool size: %d, workers: 5)\n", POOL_SIZE);

    for (i = 0; i < 5; i++) {
        threads[i] = kthread_run(worker_thread, &ids[i], "worker_%d", ids[i]);
    }

    return 0;
}
```

## 🧪 实验四：RCU机制验证

> [!tip] 目标
> 理解RCU读-复制-更新机制

### 4.1 RCU基本使用

```c
// 文件: exp4_rcu_basic.c
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/rcupdate.h>
#include <linux/slab.h>
#include <linux/kthread.h>

struct my_data {
    int value;
    char name[32];
    struct rcu_head rcu;
};

static struct my_data __rcu *global_data;

// 读取数据
static int reader_thread(void *data)
{
    struct my_data *p;
    int i;

    for (i = 0; i < 20; i++) {
        rcu_read_lock();
        p = rcu_dereference(global_data);
        pr_info("Reader: value=%d, name=%s\n", p->value, p->name);
        rcu_read_unlock();

        msleep(100);
    }

    return 0;
}

// 更新数据
static int updater_thread(void *data)
{
    struct my_data *new_data, *old_data;
    int i;

    for (i = 0; i < 5; i++) {
        new_data = kmalloc(sizeof(*new_data), GFP_KERNEL);
        new_data->value = i * 10;
        snprintf(new_data->name, sizeof(new_data->name), "version_%d", i);

        old_data = rcu_dereference(global_data);
        rcu_assign_pointer(global_data, new_data);

        pr_info("Updater: published new data (value=%d)\n", new_data->value);

        // 等待宽限期
        synchronize_rcu();
        kfree(old_data);

        msleep(500);
    }

    return 0;
}

static int __init rcu_init(void)
{
    struct task_struct *reader, *updater;
    struct my_data *initial;

    // 初始化数据
    initial = kmalloc(sizeof(*initial), GFP_KERNEL);
    initial->value = 0;
    strcpy(initial->name, "initial");
    rcu_assign_pointer(global_data, initial);

    pr_info("RCU test starting\n");

    reader = kthread_run(reader_thread, NULL, "rcu_reader");
    updater = kthread_run(updater_thread, NULL, "rcu_updater");

    return 0;
}
```

### 4.2 RCU链表操作

```c
// 文件: exp4_rcu_list.c
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/rculist.h>
#include <linux/slab.h>

struct item {
    int id;
    struct list_head list;
    struct rcu_head rcu;
};

static LIST_HEAD(item_list);
static DEFINE_SPINLOCK(list_lock);

// 添加项
void add_item(int id)
{
    struct item *new_item = kmalloc(sizeof(*new_item), GFP_KERNEL);
    new_item->id = id;

    spin_lock(&list_lock);
    list_add_rcu(&new_item->list, &item_list);
    spin_unlock(&list_lock);
}

// 删除项
void remove_item(int id)
{
    struct item *item;

    spin_lock(&list_lock);
    list_for_each_entry(item, &item_list, list) {
        if (item->id == id) {
            list_del_rcu(&item->list);
            spin_unlock(&list_lock);
            kfree_rcu(item, rcu);
            return;
        }
    }
    spin_unlock(&list_lock);
}

// 遍历项（无锁读取）
void traverse_items(void)
{
    struct item *item;

    rcu_read_lock();
    list_for_each_entry_rcu(item, &item_list, list) {
        pr_info("Item: id=%d\n", item->id);
    }
    rcu_read_unlock();
}
```

// 对比 RCU vs rwlock 的读性能

static int rcu_reader(void *data)
{
    ktime_t start, end;
    int i;
    int iterations = 10000000;

    start = ktime_get();
    for (i = 0; i < iterations; i++) {
        rcu_read_lock();
        // 读取操作
        rcu_read_unlock();
    }
    end = ktime_get();

    pr_info("RCU: %lld ns for %d reads\n",
            ktime_to_ns(ktime_sub(end, start)), iterations);

    return 0;
}

static int rwlock_reader(void *data)
{
    static DEFINE_RWLOCK(test_rwlock);

    for (i = 0; i < iterations; i++) {
        read_lock(&test_rwlock);
        // 读取操作
        read_unlock(&test_rwlock);
    }
    end = ktime_get();

    pr_info("rwlock: %lld ns for %d reads\n",
            ktime_to_ns(ktime_sub(end, start)), iterations);

    return 0;
}
