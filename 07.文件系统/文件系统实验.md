## 🧪 实验一：VFS对象分析

> [!tip] 目标
> 观察VFS核心对象：superblock、inode、dentry、file

### 1.1 查看超级块信息

```bash
# 使用dumpe2fs查看Ext4超级块
sudo dumpe2fs /dev/sda1 | head -60

# 关键字段
Filesystem volume name:   <none>
Filesystem UUID:          xxxxxxxx-xxxx-xxxx-xxxx-xxxxxxxxxxxx
Filesystem magic number:  0xEF53    # Ext4魔数
Filesystem revision #:    1 (dynamic)
Filesystem features:      has_journal, extent, huge_file, dir_nlink, extra_isize
Filesystem state:         clean
Errors behavior:          Continue
Filesystem OS type:       Linux
Inode count:              6553600
Block count:              26214400
Reserved block count:     1310720
Free blocks:              20000000
Free inodes:              6500000
First block:              0
Block size:               4096
Inode size:               256
```

### 1.2 观察inode信息

```bash
# 查看文件的inode号
ls -li /etc/passwd
# 输出: 1234567 -rw-r--r-- 1 root root 2847 Jan 1 12:00 /etc/passwd

# 查看inode详细信息
stat /etc/passwd
# File: /etc/passwd
# Size: 2847          Blocks: 8          IO Block: 4096   regular file
# Device: 801h/2049d  Inode: 1234567     Links: 1
# Access: (0644/-rw-r--r--)  Uid: (0/root)   Gid: (0/root)
# Access: 2026-03-15 10:00:00.000000000 +0800
# Modify: 2026-03-01 12:00:00.000000000 +0800
# Change: 2026-03-01 12:00:00.000000000 +0800

# 查看目录的inode
ls -lid /etc
stat /etc
```

### 1.3 观察dentry缓存

```bash
# 查看dentry缓存统计
cat /proc/sys/fs/dentry-state
# 格式: nr_dentry nr_unused age_limit want_pages pseudo_reclaim

# 示例输出: 500000 400000 45 0 0
# nr_dentry: 活跃dentry数
# nr_unused: 未使用dentry数

# 查看slab中dentry占用
cat /proc/slabinfo | grep dentry
```

### 1.4 观察打开的文件

```bash
# 查看进程打开的文件
ls -la /proc/$$/fd/
lsof -p $$

# 查看系统打开文件数
cat /proc/sys/fs/file-nr
# 格式: 已分配 未使用 最大值

# 查看文件描述符限制
ulimit -n
cat /proc/$$/limits | grep "open files"
```

## 🧪 实验三：文件系统挂载

> [!tip] 目标
> 理解挂载机制和挂载命名空间

### 3.1 基本挂载操作

```bash
# 创建镜像
dd if=/dev/zero of=mount_test.img bs=1M count=20
mkfs.ext4 mount_test.img

# 挂载
sudo mkdir -p /mnt/test1
sudo mount -o loop mount_test.img /mnt/test1

# 查看挂载信息
mount | grep test1
findmnt /mnt/test1
cat /proc/mounts | grep test1

# 在挂载点创建文件
sudo touch /mnt/test1/testfile
ls -la /mnt/test1/

# 卸载
sudo umount /mnt/test1
```

### 3.2 绑定挂载

```bash
# 创建源目录
mkdir -p /tmp/source
echo "bind mount test" > /tmp/source/file.txt

# 创建目标目录
mkdir -p /tmp/target

# 绑定挂载
sudo mount --bind /tmp/source /tmp/target

# 验证
cat /tmp/target/file.txt

# 修改验证
echo "modified" >> /tmp/target/file.txt
cat /tmp/source/file.txt   # 源文件也被修改

# 卸载
sudo umount /tmp/target
```

### 3.3 挂载选项实验

```bash
# 只读挂载
sudo mount -o ro,loop mount_test.img /mnt/test1
touch /mnt/test1/test    # 失败: Read-only file system

# 重新挂载为读写
sudo mount -o remount,rw /mnt/test1
touch /mnt/test1/test    # 成功

# noatime挂载
sudo mount -o remount,noatime /mnt/test1

# 查看挂载选项
findmnt -n -o OPTIONS /mnt/test1
```

### 3.4 观察挂载命名空间

```bash
# 查看当前进程的命名空间
ls -la /proc/self/ns/mnt

# 使用unshare创建新的挂载命名空间
sudo unshare --mount /bin/bash

# 在新命名空间中挂载
mount -t tmpfs tmpfs /mnt
touch /mnt/only_in_ns

# 另一个终端查看
# /mnt为空（不同命名空间）

# 退出命名空间
exit
```

## 🧪 实验四：页缓存观察

> [!tip] 目标
> 理解文件系统的页缓存机制

### 4.1 观察页缓存

```bash
# 清除缓存
sudo sync
echo 3 | sudo tee /proc/sys/vm/drop_caches

# 查看缓存状态
free -h
cat /proc/meminfo | grep -E "Cached|Buffers"

# 读取文件
cat /etc/passwd > /dev/null

# 再次查看缓存
free -h
cat /proc/meminfo | grep -E "Cached|Buffers"

# 查看文件的页缓存
sudo /usr/sbin/finfo /etc/passwd  # 如果安装了
# 或使用vmtouch
vmtouch /etc/passwd
```

### 4.2 使用vmtouch分析

```bash
# 安装vmtouch
sudo apt install vmtouch

# 查看文件缓存状态
vmtouch -v /etc/passwd

# 查看目录缓存状态
vmtouch -v /var/log/

# 将文件锁定在内存中
vmtouch -l /etc/passwd

# 从缓存中驱逐
vmtouch -e /etc/passwd
```

### 4.3 观察页面错误

```c
// 文件: exp4_page_fault.c
#include <stdio.h>
#include <stdlib.h>
#include <sys/mman.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <unistd.h>

int main(int argc, char *argv[])
{
    int fd;
    struct stat st;
    char *map;
    int i;

    if (argc != 2) {
        fprintf(stderr, "Usage: %s <file>\n", argv[0]);
        return 1;
    }

    fd = open(argv[1], O_RDONLY);
    if (fd < 0) {
        perror("open");
        return 1;
    }

    fstat(fd, &st);

    printf("File size: %ld bytes\n", st.st_size);

    // 使用mmap映射文件
    map = mmap(NULL, st.st_size, PROT_READ, MAP_PRIVATE, fd, 0);
    if (map == MAP_FAILED) {
        perror("mmap");
        close(fd);
        return 1;
    }

    printf("File mapped. Press Enter to access...\n");
    getchar();

    // 访问每一页
    for (i = 0; i < st.st_size; i += 4096) {
        volatile char c = map[i];
    }

    printf("Accessed all pages. Press Enter to exit...\n");
    getchar();

    munmap(map, st.st_size);
    close(fd);

    return 0;
}
```

```bash
# 编译
gcc -o exp4_page_fault exp4_page_fault.c

# 运行并观察缺页
# 另一个终端执行:
sudo perf stat -e faults ./exp4_page_fault /etc/passwd
```

### 4.4 观察写入行为


## 🧪 实验五：文件系统性能测试

> [!tip] 目标
> 比较不同文件系统的性能差异

### 5.1 使用fio测试

```bash
# 安装fio
sudo apt install fio

# 创建测试配置
cat > rand_read.fio << 'EOF'
[randread]
filename=/mnt/test1/testfile
size=1G
bs=4k
ioengine=libaio
iodepth=16
numjobs=4
rw=randread
time_based
runtime=60
group_reporting
EOF

# 运行测试
sudo fio rand_read.fio
```

### 5.2 顺序读写测试

```bash
# 使用dd测试顺序写
dd if=/dev/zero of=/mnt/test1/seq_write bs=1M count=1000 oflag=direct

# 使用dd测试顺序读
dd if=/mnt/test1/seq_write of=/dev/null bs=1M iflag=direct

# 清除缓存后测试
sudo sh -c 'echo 3 > /proc/sys/vm/drop_caches'
dd if=/mnt/test1/seq_write of=/dev/null bs=1M
```

### 5.3 小文件性能测试

```bash
# 创建大量小文件
mkdir -p /mnt/test1/small_files
time for i in {1..10000}; do
    echo "test" > /mnt/test1/small_files/file_$i
done

# 读取测试
time for i in {1..10000}; do
    cat /mnt/test1/small_files/file_$i > /dev/null
done

# 删除测试
time rm -rf /mnt/test1/small_files
```
