# 文件系统挂载

> [!info] 目标
> 理解Linux文件系统挂载的原理与实现

> [!note] 文件系统完整笔记
> 本笔记详细介绍挂载机制，总览请参考 [[Linux文件系统|MOC]]

## 挂载概述

**挂载**：将文件系统连接到目录树的过程

```
挂载前：
/ (根文件系统)
├── bin/
├── etc/
└── mnt/
    └── (空目录)

挂载后 (mount /dev/sdb1 /mnt/data)：
/ (根文件系统)
├── bin/
├── etc/
└── mnt/
    └── data/      ← 新文件系统
        ├── file1
        └── dir1/
```

## 挂载命名空间

### 概念

```c
// 每个进程可以有自己的挂载视图
struct mnt_namespace {
    struct ns_common    ns;         // 命名空间
    struct mount        *root;      // 根挂载
    struct list_head    list;       // 挂载链表
    // ...
};

// 进程的命名空间
struct task_struct {
    // ...
    struct nsproxy *nsproxy;  // 命名空间代理
};

struct nsproxy {
    struct mnt_namespace *mnt_ns;  // 挂载命名空间
    // ...
};
```

### 共享子树

```
挂载传播类型：

shared   : 共享挂载，传播到其他命名空间
slave    : 从属挂载，只接收不传播
private  : 私有挂载，不传播
unbindable: 不可绑定，不能被bind mount
```

```bash
# 设置传播类型
mount --make-shared /mnt
mount --make-slave /mnt
mount --make-private /mnt
mount --make-unbindable /mnt
```

## 核心数据结构

### mount结构

```c
// fs/mount.h
struct mount {
    struct hlist_node mnt_hash;         // 哈希链表
    struct mount *mnt_parent;           // 父挂载

    struct vfsmount mnt;                // VFS挂载信息
    struct list_head mnt_child;         // 子挂载链表
    struct list_head mnt_mounts;        // 挂载列表
    struct mnt_namespace *mnt_ns;       // 所属命名空间
    // ...
};

struct vfsmount {
    struct dentry *mnt_root;            // 根dentry
    struct super_block *mnt_sb;         // 超级块
    int mnt_flags;                      // 挂载标志
    // ...
};
```

## 挂载流程

### mount系统调用

```c
// fs/namespace.c
SYSCALL_DEFINE5(mount, const char __user *, dev_name,
                const char __user *, dir_name,
                const char __user *, type, unsigned long, flags,
                const void __user *, data)
{
    return ksys_mount(dev_name, dir_name, type, flags, data);
}
```

```
mount("/dev/sda1", "/mnt", "ext4", 0, NULL)
```

### Ext4挂载示例

```c
// fs/ext4/super.c
static struct dentry *ext4_mount(struct file_system_type *fs_type, int flags,
                                  const char *dev_name, void *data)
{
    return mount_bdev(fs_type, flags, dev_name, data, ext4_fill_super);
}

static int ext4_fill_super(struct super_block *sb, void *data, int silent)
{
    struct ext4_sb_info *sbi;
    struct buffer_head *bh;
    struct ext4_super_block *es;

    // 1. 分配sbi
    sbi = kzalloc(sizeof(*sbi), GFP_KERNEL);
    sb->s_fs_info = sbi;

    // 2. 读取超级块
    bh = sb_bread(sb, 1);
    es = (struct ext4_super_block *) bh->b_data;

    // 3. 检查魔数
    if (le16_to_cpu(es->s_magic) != EXT4_SUPER_MAGIC) {
        // 不是Ext4文件系统
        return -EINVAL;
    }

    // 4. 设置VFS超级块
    sb->s_blocksize = 1 << (le32_to_cpu(es->s_log_block_size) + 10);
    sb->s_maxbytes = ext4_max_size(sb->s_blocksize_bits);
```

## 挂载选项

### 常用挂载选项

```bash
# 基本选项
mount -t ext4 /dev/sda1 /mnt

# 指定选项
mount -t ext4 -o noatime,nodiratime /dev/sda1 /mnt

# 多个选项
mount -o rw,noatime,data=ordered /dev/sda1 /mnt
```

### Ext4常用选项

| 选项 | 说明 |
|------|------|
| noatime | 不更新访问时间 |
| nodiratime | 目录不更新访问时间 |
| data=journal | 日志模式 |
| data=ordered | 有序模式（默认） |
| data=writeback | 写回模式 |
| journal_dev | 指定日志设备 |
| barrier | 启用写入屏障（默认） |
| nobarrier | 禁用写入屏障 |
| discard | 启用TRIM |
| errors=remount-ro | 错误时重新挂载为只读 |
| errors=continue | 错误时继续 |
| errors=panic | 错误时内核panic |

### 查看挂载选项

```bash
# 方法1
mount | grep /mnt

# 方法2
cat /proc/mounts | grep /mnt

# 方法3
findmnt /mnt

# 方法4
cat /proc/self/mountinfo
```

## 特殊挂载类型

### Bind Mount

```bash
# 绑定挂载：将目录挂载到另一个位置
mount --bind /source /dest

# 递归绑定
mount --rbind /source /dest

# 用途：
# - chroot环境
# - 容器文件系统
# - 系统维护
```

### Move Mount

```bash
# 移动挂载点
mount --move /old/location /new/location
```

### Remount

```bash
# 重新挂载，修改选项
mount -o remount,rw /mnt
mount -o remount,ro /mnt

# 常用：将只读改为读写
mount -o remount,rw /
```

## 文件系统类型注册

### 注册文件系统

```c
// 定义文件系统类型
static struct file_system_type ext4_fs_type = {
    .owner      = THIS_MODULE,
    .name       = "ext4",
    .mount      = ext4_mount,
    .kill_sb    = kill_block_super,
    .fs_flags   = FS_REQUIRES_DEV,
};

// 注册
static int __init ext4_init_fs(void)
{
    return register_filesystem(&ext4_fs_type);
}

// 取消注册
static void __exit ext4_exit_fs(void)
{
    unregister_filesystem(&ext4_fs_type);
}

module_init(ext4_init_fs)
module_exit(ext4_exit_fs)
```

## 自动挂载

### /etc/fstab

```bash
# 格式
# 设备      挂载点   类型    选项         dump fsck
/dev/sda1   /        ext4    defaults     1     1
/dev/sda2   /home    ext4    defaults     0     2
/dev/sda3   swap     swap    defaults     0     0
proc        /proc    proc    defaults     0     0
sysfs       /sys     sysfs   defaults     0     0

# 选项说明
defaults = rw, suid, dev, exec, auto, nouser, async

# 挂载所有
mount -a
```

### autofs

```bash
# 安装autofs
apt install autofs

# 配置 /etc/auto.master
/mnt/auto /etc/auto.misc

# 配置 /etc/auto.misc
data    -fstype=ext4    :/dev/sdb1
nfs     -fstype=nfs     server:/export

# 启动服务
systemctl start autofs
```

## 相关命令

```bash
# 查看挂载
mount                    # 所有挂载
findmnt                  # 树形显示
cat /proc/mounts         # 内核视图

# 挂载
mount /dev/sda1 /mnt     # 基本挂载

mount -o loop image.iso /mnt  # 挂载镜像

# 卸载
umount /mnt              # 卸载
umount -l /mnt           # 懒卸载（忙时）

```
