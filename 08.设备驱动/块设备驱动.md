# 块设备驱动

> [!info] 目标
> 掌握Linux块设备驱动开发

> [!note] 驱动完整笔记
> 本笔记详细介绍块设备驱动，总览请参考 [[Linux设备驱动|MOC]]

## 块设备概述

### 特点

- 按**块**（通常512字节-4KB）访问
- 支持**随机访问**（可跳转到任意位置）
- 有**缓存机制**（页缓存、块缓存）
- 通过**请求队列**处理I/O

### 与字符设备的区别

| 特性 | 字符设备 | 块设备 |
|------|----------|--------|
| 访问单位 | 字节 | 块（扇区） |
| 随机访问 | 可选 | 必须支持 |
| 缓存 | 应用层 | 内核页缓存 |
| 注册方式 | cdev | gendisk |
| I/O模型 | 直接读写 | 请求队列 |

## 核心数据结构

### gendisk (通用磁盘)

```c
// include/linux/genhd.h
struct gendisk {
    int major;                      // 主设备号

    struct block_device_operations *fops;  // 操作函数表
    struct request_queue *queue;    // 请求队列
```

## 块设备注册流程

### 传统方式（无blk-mq）

```
1. alloc_disk()          分配gendisk
2. init_request_queue()  初始化请求队列
3. 设置gendisk属性
4. add_disk()            添加磁盘
```

### 现代方式（blk-mq）

```c
// 1. 定义硬件队列数量和深度
#define NUM_HW_QUEUES 1
#define QUEUE_DEPTH 128

// 2. 定义blk_mq_ops
static const struct blk_mq_ops my_mq_ops = {
    .queue_rq = my_queue_rq,      // 处理请求
    .init_request = my_init_request,
    .exit_request = my_exit_request,
    .map_queue = blk_mq_map_queue,
};

// 3. 初始化tag_set
static struct blk_mq_tag_set my_tag_set;

static int my_init_tag_set(void)
{
    my_tag_set.ops = &my_mq_ops;
    my_tag_set.nr_hw_queues = NUM_HW_QUEUES;
    my_tag_set.queue_depth = QUEUE_DEPTH;
    my_tag_set.numa_node = NUMA_NO_NODE;
    my_tag_set.cmd_size = sizeof(struct my_cmd);
    my_tag_set.flags = BLK_MQ_F_SHOULD_MERGE;

    return blk_mq_alloc_tag_set(&my_tag_set);
}

// 4. 创建gendisk
static int my_add_disk(void)
{
    struct gendisk *disk;

    disk = alloc_disk(1);  // 1个分区
    if (!disk)
        return -ENOMEM;

    disk->major = my_major;
    disk->first_minor = 0;
    disk->fops = &my_fops;
    sprintf(disk->disk_name, "mydisk");

    // 创建请求队列
    disk->queue = blk_mq_init_queue(&my_tag_set);
    if (IS_ERR(disk->queue)) {
        put_disk(disk);
        return PTR_ERR(disk->queue);
    }

    set_capacity(disk, MY_DISK_SIZE >> 9);  // 扇区数

    add_disk(disk);

    return 0;
}
```

## 请求处理

### queue_rq回调

```c
static blk_status_t my_queue_rq(struct blk_mq_hw_ctx *hctx,
                                 const struct blk_mq_queue_data *bd)
{
    struct request *rq = bd->rq;
    struct my_device *dev = rq->q->queuedata;

    // 开始请求
    blk_mq_start_request(rq);

    // 处理请求
    switch (req_op(rq)) {
    case REQ_OP_READ:
        my_do_read(dev, rq);
        break;
    case REQ_OP_WRITE:
        my_do_write(dev, rq);
        break;
    case REQ_OP_FLUSH:
        my_do_flush(dev, rq);
        break;
    case REQ_OP_DISCARD:
        my_do_discard(dev, rq);
        break;
    default:
        blk_mq_end_request(rq, BLK_STS_IOERR);
        return BLK_STS_OK;
    }

    return BLK_STS_OK;
}
```

### 完成请求

```c
// 同步完成
blk_mq_end_request(rq, BLK_STS_OK);

// 异步完成（在中断中）
blk_mq_complete_request(rq);

// 完成回调
static void my_complete_rq(struct request *rq)
{
    blk_mq_end_request(rq, BLK_STS_OK);
}
```

## 简单块设备示例

```c
// 文件: simple_blkdev.c
#include <linux/module.h>
#include <linux/blkdev.h>
#include <linux/vmalloc.h>

#define SECTOR_SIZE 512
#define DISK_SIZE (16 * 1024 * 1024)  // 16MB

static int my_major;
static struct gendisk *my_disk;
static struct blk_mq_tag_set my_tag_set;
static struct request_queue *my_queue;
static void *my_data;  // 模拟存储

static blk_status_t my_queue_rq(struct blk_mq_hw_ctx *hctx,
                                 const struct blk_mq_queue_data *bd)
{
    struct request *rq = bd->rq;
    struct bio_vec bvec;
    struct bvec_iter iter;
    sector_t sector;
    void *addr;

    blk_mq_start_request(rq);

    sector = blk_rq_pos(rq);

    // 遍历bio中的每个segment
    rq_for_each_segment(bvec, rq, iter) {
        addr = page_address(bvec.bv_page) + bvec.bv_offset;

        if (rq_data_dir(rq) == WRITE) {
            memcpy(my_data + (sector << 9), addr, bvec.bv_len);
        } else {
            memcpy(addr, my_data + (sector << 9), bvec.bv_len);
        }

        sector += bvec.bv_len >> 9;
    }

    blk_mq_end_request(rq, BLK_STS_OK);
    return BLK_STS_OK;
}

static const struct blk_mq_ops my_mq_ops = {
    .queue_rq = my_queue_rq,
};

static const struct block_device_operations my_fops = {
    .owner = THIS_MODULE,
};

static int __init my_init(void)
{
    int ret;

    // 分配存储空间
    my_data = vmalloc(DISK_SIZE);
    if (!my_data)
        return -ENOMEM;

    // 动态分配主设备号
    my_major = register_blkdev(0, "myblkdev");
    if (my_major < 0) {
        vfree(my_data);
        return my_major;
    }

    // 初始化tag_set
    memset(&my_tag_set, 0, sizeof(my_tag_set));
    my_tag_set.ops = &my_mq_ops;
    my_tag_set.nr_hw_queues = 1;
    my_tag_set.queue_depth = 128;
    my_tag_set.numa_node = NUMA_NO_NODE;
    my_tag_set.cmd_size = 0;
    my_tag_set.flags = BLK_MQ_F_SHOULD_MERGE;

    ret = blk_mq_alloc_tag_set(&my_tag_set);
    if (ret) {
        unregister_blkdev(my_major, "myblkdev");
        vfree(my_data);
        return ret;
    }

    // 分配disk
    my_disk = alloc_disk(1);
    if (!my_disk) {
        blk_mq_free_tag_set(&my_tag_set);
        unregister_blkdev(my_major, "myblkdev");
        vfree(my_data);
        return -ENOMEM;
    }

    // 创建队列
    my_queue = blk_mq_init_queue(&my_tag_set);
    if (IS_ERR(my_queue)) {
        put_disk(my_disk);
        blk_mq_free_tag_set(&my_tag_set);
        unregister_blkdev(my_major, "myblkdev");
        vfree(my_data);
        return PTR_ERR(my_queue);
    }

    // 设置disk属性
    my_disk->major = my_major;
    my_disk->first_minor = 0;
    my_disk->fops = &my_fops;
    my_disk->queue = my_queue;
    sprintf(my_disk->disk_name, "myblkdev");
    set_capacity(my_disk, DISK_SIZE >> 9);

    // 添加disk
    add_disk(my_disk);

    pr_info("Block device registered: /dev/%s\n", my_disk->disk_name);
    return 0;
}

static void __exit my_exit(void)
{
    del_gendisk(my_disk);
    blk_cleanup_queue(my_queue);
    put_disk(my_disk);
    blk_mq_free_tag_set(&my_tag_set);
    unregister_blkdev(my_major, "myblkdev");
    vfree(my_data);

    pr_info("Block device unregistered\n");
}

module_init(my_init);
module_exit(my_exit);
MODULE_LICENSE("GPL");
```

## I/O调度器

### 常见调度器

| 调度器 | 说明 | 适用场景 |
|--------|------|----------|
| none/mq-deadline | 无调度/简单调度 | NVMe SSD |
| deadline | 截止时间调度 | SSD |
| cfq | 完全公平队列 | HDD |
| bfq | 预算公平队列 | 桌面HDD |

### 设置调度器

```bash
# 查看当前调度器
cat /sys/block/sda/queue/scheduler

# 设置调度器
echo deadline > /sys/block/sda/queue/scheduler
```
