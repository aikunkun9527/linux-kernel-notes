## 字符设备概述

字符设备特点：
- 按字节流访问
- 支持随机访问（可选）
- 如：串口、键盘、鼠标

## 设备号

### 主设备号与次设备号

```c
#include <linux/kdev_t.h>

dev_t dev;  // 32位：主设备号(12位) + 次设备号(20位)

// 获取主次设备号
MAJOR(dev);  // 主设备号
MINOR(dev);  // 次设备号

// 构建设备号
MKDEV(major, minor);
```

### 分配设备号

```c
// 静态分配（需确保未使用）
int register_chrdev_region(dev_t from, unsigned count, const char *name);

// 动态分配（推荐）
int alloc_chrdev_region(dev_t *dev, unsigned baseminor,
                        unsigned count, const char *name);

// 释放
void unregister_chrdev_region(dev_t from, unsigned count);
```

## cdev结构体

```c
#include <linux/cdev.h>

struct cdev {
    struct kobject kobj;
    struct module *owner;
    const struct file_operations *ops;
    struct list_head list;
    dev_t dev;
    unsigned int count;
};

// 初始化
void cdev_init(struct cdev *cdev, const struct file_operations *fops);

// 添加到系统
int cdev_add(struct cdev *p, dev_t dev, unsigned count);

// 删除
void cdev_del(struct cdev *p);
```

## 完整驱动示例

```c
#include <linux/module.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/uaccess.h>

#define DEVICE_NAME "mychardev"
#define BUF_LEN 1024

static int major;
static struct cdev my_cdev;
static char device_buffer[BUF_LEN];

// 打开
static int my_open(struct inode *inode, struct file *file)
{
    pr_info("Device opened\n");
    return 0;
}

// 释放
static int my_release(struct inode *inode, struct file *file)
{
    pr_info("Device closed\n");
    return 0;
}

// 读
static ssize_t my_read(struct file *file, char __user *buf,
                       size_t count, loff_t *offset)
{
    if (*offset >= BUF_LEN)
        return 0;
    if (*offset + count > BUF_LEN)
        count = BUF_LEN - *offset;

    if (copy_to_user(buf, device_buffer + *offset, count))
        return -EFAULT;

    *offset += count;
    return count;
}

// 写
static ssize_t my_write(struct file *file, const char __user *buf,
                        size_t count, loff_t *offset)
{
    if (*offset >= BUF_LEN)
        return -ENOSPC;
    if (*offset + count > BUF_LEN)
        count = BUF_LEN - *offset;

    if (copy_from_user(device_buffer + *offset, buf, count))
        return -EFAULT;

    *offset += count;
    return count;
}

// 文件操作
static const struct file_operations fops = {
    .owner = THIS_MODULE,
    .open = my_open,
    .release = my_release,
    .read = my_read,
    .write = my_write,
};

static int __init my_init(void)
{
    dev_t dev;
    int ret;

    // 动态分配设备号
    ret = alloc_chrdev_region(&dev, 0, 1, DEVICE_NAME);
    if (ret < 0)
        return ret;
    major = MAJOR(dev);

    // 初始化cdev
    cdev_init(&my_cdev, &fops);
    my_cdev.owner = THIS_MODULE;

    // 添加设备
    ret = cdev_add(&my_cdev, dev, 1);
    if (ret)
        goto fail;

    pr_info("Device registered with major %d\n", major);
    return 0;

fail:
    unregister_chrdev_region(dev, 1);
    return ret;
}

static void __exit my_exit(void)
{
    cdev_del(&my_cdev);
    unregister_chrdev_region(MKDEV(major, 0), 1);
    pr_info("Device unregistered\n");
}

module_init(my_init);
module_exit(my_exit);
MODULE_LICENSE("GPL");
```

## 自动创建设备节点

```c
#include <linux/device.h>

static struct class *my_class;
static struct device *my_device;

// 在init中
my_class = class_create(THIS_MODULE, "my_class");
my_device = device_create(my_class, NULL, dev, NULL, "mychardev");

// 在exit中
device_destroy(my_class, dev);
class_destroy(my_class);
```
