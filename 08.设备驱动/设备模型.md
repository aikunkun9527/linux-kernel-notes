# Linux设备模型

> [!info] 目标
> 理解Linux设备模型的架构与sysfs文件系统

> [!note] 驱动完整笔记
> 本笔记详细介绍设备模型，总览请参考 [[Linux设备驱动|MOC]]

## 设备模型概述

### 为什么需要设备模型

```
设备模型解决的问题：

1. 设备发现
   - 系统中有哪些设备？
   - 设备连接在哪里？

2. 驱动绑定
   - 哪个驱动管理哪个设备？
   - 如何自动加载驱动？

3. 电源管理
   - 设备的电源状态？
   - 如何协调休眠/唤醒？

4. 用户空间接口
   - 如何查看设备信息？
   - 如何与设备交互？
```

### 核心概念

```
┌─────────────────────────────────────────────────────────────┐
│                    设备模型层次结构                          │
```

## kobject

### 结构定义

```c
// include/linux/kobject.h
struct kobject {
    const char              *name;      // 名称
    struct list_head        entry;      // 链入父kset的链表
    struct kobject          *parent;    // 父对象
    struct kset             *kset;      // 所属kset
    const struct kobj_type  *ktype;     // 类型
    struct kernfs_node      *sd;        // sysfs目录项
    struct kref             kref;       // 引用计数

#ifdef CONFIG_DEBUG_KOBJECT_RELEASE
    struct delayed_work     release;
#endif
    unsigned int            state_initialized:1;
    unsigned int            state_in_sysfs:1;
    unsigned int            state_add_uevent_sent:1;
    unsigned int            state_remove_uevent_sent:1;
    unsigned int            uevent_suppress:1;
};
```

### kobject操作

```c
// 初始化
void kobject_init(struct kobject *kobj, const struct kobj_type *ktype);

// 初始化并添加到sysfs
int kobject_add(struct kobject *kobj, struct kobject *parent,
                const char *fmt, ...);

// 初始化、添加（组合）
int kobject_init_and_add(struct kobject *kobj, const struct kobj_type *ktype,
                         struct kobject *parent, const char *fmt, ...);

// 增加引用计数
struct kobject *kobject_get(struct kobject *kobj);

// 减少引用计数，为0时释放
void kobject_put(struct kobject *kobj);

// 创建并添加
struct kobject *kobject_create_and_add(const char *name, struct kobject *parent);

// 从sysfs移除
void kobject_del(struct kobject *kobj);
```

### kobj_type

```c
struct kobj_type {
    void (*release)(struct kobject *kobj);  // 释放回调
    const struct sysfs_ops *sysfs_ops;      // sysfs操作
    const struct attribute **default_attrs; // 默认属性
    const struct kobj_ns_type_operations *(*child_ns_type)(struct kobject *kobj);
    const void *(*namespace)(struct kobject *kobj);
    void (*get_ownership)(struct kobject *kobj, kuid_t *uid, kgid_t *gid);
};

// sysfs操作
struct sysfs_ops {
    ssize_t (*show)(struct kobject *kobj, struct attribute *attr, char *buf);
    ssize_t (*store)(struct kobject *kobj, struct attribute *attr,
                     const char *buf, size_t count);
};
```

## kset

### 结构定义

```c
struct kset {
    struct list_head        list;       // 包含的kobject链表
    spinlock_t              list_lock;  // 保护链表的锁
    struct kobject          kobj;       // 嵌入的kobject
    const struct kset_uevent_ops *uevent_ops;  // uevent操作
};
```

### kset操作

```c
// 初始化
void kset_init(struct kset *kset);

// 注册/注销
int kset_register(struct kset *kset);
void kset_unregister(struct kset *kset);

// 创建并添加
struct kset *kset_create_and_add(const char *name,
                                  const struct kset_uevent_ops *uevent_ops,
                                  struct kobject *parent_kobj);
```

### kset与kobject关系

```
                    kset
┌─────────────────────────────────────────┐
```

## device

### 结构定义

```c
// include/linux/device.h
struct device {
    struct device       *parent;        // 父设备

    struct device_private *p;           // 私有数据

    struct kobject kobj;                // 嵌入的kobject
    const char      *init_name;         // 初始名称
    const struct device_type *type;     // 设备类型

    struct mutex      mutex;            // 互斥锁

    struct bus_type   *bus;             // 所属总线
    struct device_driver *driver;       // 绑定的驱动

    void              *platform_data;   // 平台数据
    void              *driver_data;     // 驱动数据

    struct dev_pm_info power;           // 电源管理
    struct dev_pm_domain *pm_domain;    // 电源域

    struct device_node *of_node;        // 设备树节点

    dev_t             devt;             // 设备号
    u32               id;               // 设备ID

    spinlock_t        devres_lock;      // 资源锁
    struct list_head  devres_head;      // 资源链表

    struct class      *class;           // 所属类
    const struct attribute_group **groups;

    void    (*release)(struct device *dev);  // 释放回调
    // ...
};
```

## device_driver

### 结构定义

```c
struct device_driver {
    const char              *name;       // 驱动名称
    struct bus_type         *bus;       // 所属总线

    struct module           *owner;     // 所属模块
    const char              *mod_name;  // 模块名

    bool                    suppress_bind_attrs;
    bool                    probe_type;

    const struct of_device_id  *of_match_table;  // 设备树匹配
    const struct acpi_device_id *acpi_match_table;

    int (*probe)(struct device *dev);   // 探测
    int (*remove)(struct device *dev);  // 移除
    void (*shutdown)(struct device *dev);
    int (*suspend)(struct device *dev, pm_message_t state);
    int (*resume)(struct device *dev);

    const struct attribute_group **groups;

    const struct dev_pm_ops *pm;        // 电源管理操作

    struct driver_private *p;           // 私有数据
};
```

## platform_driver

### 平台设备驱动

```c
#include <linux/platform_device.h>

struct platform_driver {
    int (*probe)(struct platform_device *);
    int (*remove)(struct platform_device *);
    void (*shutdown)(struct platform_device *);
    int (*suspend)(struct platform_device *, pm_message_t state);
    int (*resume)(struct platform_device *);
    struct device_driver driver;
    const struct platform_device_id *id_table;
    bool prevent_deferred_probe;
};

// 注册
int platform_driver_register(struct platform_driver *);
void platform_driver_unregister(struct platform_driver *);

// 简化注册（不支持热插拔）
module_platform_driver(my_driver);
```

### 平台设备驱动示例

```c
#include <linux/module.h>
#include <linux/platform_device.h>
#include <linux/mod_devicetable.h>

static int my_probe(struct platform_device *pdev)
{
    pr_info("Device probed: %s\n", dev_name(&pdev->dev));
    return 0;
}

static int my_remove(struct platform_device *pdev)
{
    pr_info("Device removed: %s\n", dev_name(&pdev->dev));
    return 0;
}

static const struct of_device_id my_of_match[] = {
    { .compatible = "vendor,mydevice", },
    { }
};
MODULE_DEVICE_TABLE(of, my_of_match);

static struct platform_driver my_driver = {
    .probe = my_probe,
    .remove = my_remove,
    .driver = {
        .name = "mydevice",
        .of_match_table = my_of_match,
    },
};

module_platform_driver(my_driver);
MODULE_LICENSE("GPL");
```

## sysfs

### 目录结构

```
/sys/
├── block/         # 块设备
│   ├── sda/
│   └── sdb/
├── bus/           # 总线
│   ├── pci/
│   ├── usb/
│   └── platform/
├── class/         # 设备类
│   ├── net/
│   ├── input/
│   └── tty/
├── devices/       # 设备（按拓扑）
│   ├── platform/
│   ├── pci0000:00/
│   └── virtual/
├── dev/           # 设备节点
│   ├── block/
│   └── char/
├── firmware/      # 固件
├── fs/            # 文件系统
├── kernel/        # 内核参数
├── module/        # 模块信息
└── power/         # 电源管理
```

### 创建sysfs属性

```c
// 定义属性
static ssize_t my_attr_show(struct device *dev,
                            struct device_attribute *attr, char *buf)
{
    return sprintf(buf, "%d\n", my_value);
}

static ssize_t my_attr_store(struct device *dev,
                             struct device_attribute *attr,
                             const char *buf, size_t count)
{
    int ret;
    ret = kstrtoint(buf, 10, &my_value);
    if (ret)
        return ret;
    return count;
}

static DEVICE_ATTR(my_attr, 0644, my_attr_show, my_attr_store);

// 创建
device_create_file(dev, &dev_attr_my_attr);

// 删除
device_remove_file(dev, &dev_attr_my_attr);

// 属性组
static struct attribute *my_attrs[] = {
    &dev_attr_my_attr.attr,
    NULL,
};

static const struct attribute_group my_group = {
    .attrs = my_attrs,
};
```

## 自动创建设备节点

### 使用设备类

```c
#include <linux/device.h>

static struct class *my_class;
static int my_major;

// 初始化
my_class = class_create(THIS_MODULE, "my_class");

// 创建设备
struct device *dev = device_create(my_class, NULL, MKDEV(my_major, 0),
                                   NULL, "mydevice");

// 清理
device_destroy(my_class, MKDEV(my_major, 0));
class_destroy(my_class);
```

### udev规则

```bash
# /etc/udev/rules.d/99-mydevice.rules

# 根据属性创建设备节点
KERNEL=="mydevice", MODE="0666"

# 创建符号链接
KERNEL=="mydevice", SYMLINK+="mydev"

# 运行脚本
KERNEL=="mydevice", RUN+="/bin/echo 'device added'"
```
