### 1.1 基本字符设备驱动

```c
// 文件: exp1_chardev/mychardev.c
#include <linux/module.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/device.h>
#include <linux/uaccess.h>

#define DEVICE_NAME "mychardev"
#define BUF_LEN 1024
#define NUM_DEVICES 2

static int my_major;
static struct cdev my_cdev;
static struct class *my_class;
static char device_buffer[NUM_DEVICES][BUF_LEN];
static int open_count[NUM_DEVICES];

static int my_open(struct inode *inode, struct file *file)
{
    int minor = iminor(inode);

    if (minor >= NUM_DEVICES)
        return -ENODEV;

    file->private_data = (void *)(long)minor;
    open_count[minor]++;

    pr_info("Device %d opened, count=%d\n", minor, open_count[minor]);
    return 0;
}

static int my_release(struct inode *inode, struct file *file)
{
    int minor = (long)file->private_data;
    open_count[minor]--;

    pr_info("Device %d closed, count=%d\n", minor, open_count[minor]);
    return 0;
}

static ssize_t my_read(struct file *file, char __user *buf,
                       size_t count, loff_t *offset)
{
    int minor = (long)file->private_data;
    char *buffer = device_buffer[minor];

    if (*offset >= BUF_LEN)
        return 0;

    if (*offset + count > BUF_LEN)
        count = BUF_LEN - *offset;

    if (copy_to_user(buf, buffer + *offset, count))
        return -EFAULT;

    *offset += count;
    pr_info("Device %d: read %zu bytes\n", minor, count);
    return count;
}

static ssize_t my_write(struct file *file, const char __user *buf,
                        size_t count, loff_t *offset)
{
    int minor = (long)file->private_data;
    char *buffer = device_buffer[minor];

    if (*offset >= BUF_LEN)
        return -ENOSPC;

    if (*offset + count > BUF_LEN)
        count = BUF_LEN - *offset;

    if (copy_from_user(buffer + *offset, buf, count))
        return -EFAULT;

    *offset += count;
    pr_info("Device %d: write %zu bytes\n", minor, count);
    return count;
}

static loff_t my_llseek(struct file *file, loff_t offset, int whence)
{
    loff_t new_offset;

    switch (whence) {
    case SEEK_SET:
        new_offset = offset;
        break;
    case SEEK_CUR:
        new_offset = file->f_pos + offset;
        break;
    case SEEK_END:
        new_offset = BUF_LEN + offset;
        break;
    default:
        return -EINVAL;
    }

    if (new_offset < 0 || new_offset > BUF_LEN)
        return -EINVAL;

    file->f_pos = new_offset;
    return new_offset;
}

static const struct file_operations fops = {
    .owner = THIS_MODULE,
    .open = my_open,
    .release = my_release,
    .read = my_read,
    .write = my_write,
    .llseek = my_llseek,
};

static int __init my_init(void)
{
    dev_t dev;
    int ret, i;

    // 初始化缓冲区
    for (i = 0; i < NUM_DEVICES; i++) {
        memset(device_buffer[i], 0, BUF_LEN);
        open_count[i] = 0;
    }

    // 动态分配设备号
    ret = alloc_chrdev_region(&dev, 0, NUM_DEVICES, DEVICE_NAME);
    if (ret < 0) {
        pr_err("Failed to allocate device number\n");
        return ret;
    }
    my_major = MAJOR(dev);

    // 初始化cdev
    cdev_init(&my_cdev, &fops);
    my_cdev.owner = THIS_MODULE;

    ret = cdev_add(&my_cdev, dev, NUM_DEVICES);
    if (ret) {
        pr_err("Failed to add cdev\n");
        goto fail_cdev;
    }

    // 创建设备类
    my_class = class_create(THIS_MODULE, DEVICE_NAME);
    if (IS_ERR(my_class)) {
        pr_err("Failed to create class\n");
        ret = PTR_ERR(my_class);
        goto fail_class;
    }

    // 创建设备节点
    for (i = 0; i < NUM_DEVICES; i++) {
        device_create(my_class, NULL, MKDEV(my_major, i), NULL,
                      "mychardev%d", i);
    }

    pr_info("Driver loaded: major=%d, devices=%d\n", my_major, NUM_DEVICES);
    return 0;

fail_class:
    cdev_del(&my_cdev);
fail_cdev:
    unregister_chrdev_region(dev, NUM_DEVICES);
    return ret;
}

static void __exit my_exit(void)
{
    int i;

    for (i = 0; i < NUM_DEVICES; i++) {
        device_destroy(my_class, MKDEV(my_major, i));
    }
    class_destroy(my_class);

    cdev_del(&my_cdev);
    unregister_chrdev_region(MKDEV(my_major, 0), NUM_DEVICES);

    pr_info("Driver unloaded\n");
}

module_init(my_init);
module_exit(my_exit);
MODULE_LICENSE("GPL");
MODULE_AUTHOR("Student");
MODULE_DESCRIPTION("Simple character device driver");
```

// 文件: exp1_chardev/test.c
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>

    printf("Reading data...\n");

    printf("Read %zd bytes: %s\n", ret, buf);

    close(fd);
    return 0;

### 2.1 ioctl实现

```c
// 文件: exp2_ioctl/myioctl.c
#include <linux/module.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/device.h>
#include <linux/uaccess.h>
#include <linux/slab.h>

#define DEVICE_NAME "myioctl"

// ioctl命令定义
#define MY_MAGIC 'M'
#define MY_IOCTL_GET_VALUE    _IOR(MY_MAGIC, 1, int)
#define MY_IOCTL_SET_VALUE    _IOW(MY_MAGIC, 2, int)
#define MY_IOCTL_RESET        _IO(MY_MAGIC, 3)
#define MY_IOCTL_EXCHANGE     _IOWR(MY_MAGIC, 4, int)

static int my_major;
static struct cdev my_cdev;
static struct class *my_class;
static int device_value = 100;

static long my_ioctl(struct file *file, unsigned int cmd, unsigned long arg)
{
    int ret = 0;
    int value;

    switch (cmd) {
    case MY_IOCTL_GET_VALUE:
        ret = put_user(device_value, (int __user *)arg);
        pr_info("GET_VALUE: %d\n", device_value);
        break;

    case MY_IOCTL_SET_VALUE:
        ret = get_user(value, (int __user *)arg);
        if (ret == 0) {
            device_value = value;
            pr_info("SET_VALUE: %d\n", device_value);
        }
        break;

    case MY_IOCTL_RESET:
        device_value = 100;
        pr_info("RESET: %d\n", device_value);
        break;

    case MY_IOCTL_EXCHANGE:
        ret = get_user(value, (int __user *)arg);
        if (ret == 0) {
            int old = device_value;
            device_value = value;
            ret = put_user(old, (int __user *)arg);
            pr_info("EXCHANGE: old=%d, new=%d\n", old, device_value);
        }
        break;

    default:
        pr_warn("Unknown ioctl: 0x%x\n", cmd);
        return -ENOTTY;
    }

    return ret;
}

static const struct file_operations fops = {
    .owner = THIS_MODULE,
    .unlocked_ioctl = my_ioctl,
};

static int __init my_init(void)
{
    dev_t dev;
    int ret;

    ret = alloc_chrdev_region(&dev, 0, 1, DEVICE_NAME);
    if (ret < 0)
        return ret;
    my_major = MAJOR(dev);

    cdev_init(&my_cdev, &fops);
    my_cdev.owner = THIS_MODULE;
    cdev_add(&my_cdev, dev, 1);

    my_class = class_create(THIS_MODULE, DEVICE_NAME);
    device_create(my_class, NULL, dev, NULL, DEVICE_NAME);

    pr_info("ioctl device registered: /dev/%s\n", DEVICE_NAME);
    return 0;
}

static void __exit my_exit(void)
{
    device_destroy(my_class, MKDEV(my_major, 0));
    class_destroy(my_class);
    cdev_del(&my_cdev);
    unregister_chrdev_region(MKDEV(my_major, 0), 1);
    pr_info("ioctl device unregistered\n");
}

module_init(my_init);
module_exit(my_exit);
MODULE_LICENSE("GPL");
```

### 2.2 poll实现

```c
// 文件: exp2_poll/mypoll.c (关键部分)
#include <linux/poll.h>
#include <linux/wait.h>

static DECLARE_WAIT_QUEUE_HEAD(read_queue);
static bool data_available = false;
static char data_buffer[128];
static size_t data_len = 0;

static __poll_t my_poll(struct file *file, poll_table *wait)
{
    __poll_t mask = 0;

    poll_wait(file, &read_queue, wait);

    if (data_available)
        mask |= POLLIN | POLLRDNORM;  // 可读

    mask |= POLLOUT | POLLWRNORM;      // 总是可写

    return mask;
}

static ssize_t my_read(struct file *file, char __user *buf,
                       size_t count, loff_t *offset)
{
    if (!data_available) {
        if (file->f_flags & O_NONBLOCK)
            return -EAGAIN;
        if (wait_event_interruptible(read_queue, data_available))
            return -ERESTARTSYS;
    }

    if (count > data_len)
        count = data_len;

    if (copy_to_user(buf, data_buffer, count))
        return -EFAULT;

    data_available = false;
    data_len = 0;

    return count;
}

static ssize_t my_write(struct file *file, const char __user *buf,
                        size_t count, loff_t *offset)
{
    if (count > sizeof(data_buffer))
        count = sizeof(data_buffer);

    if (copy_from_user(data_buffer, buf, count))
        return -EFAULT;

    data_len = count;
    data_available = true;

    wake_up_interruptible(&read_queue);

    return count;
}

static const struct file_operations fops = {
    .owner = THIS_MODULE,
    .poll = my_poll,
    .read = my_read,
    .write = my_write,
};
```

### 3.1 按键中断驱动

```c
// 文件: exp3_irq/button_irq.c
#include <linux/module.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/device.h>
#include <linux/interrupt.h>
#include <linux/sched/signal.h>
#include <linux/poll.h>
#include <linux/gpio.h>

#define DEVICE_NAME "button"

static int my_major;
static struct cdev my_cdev;
static struct class *my_class;

static unsigned int gpio_pin = 17;  // GPIO 17
static int irq_num;

static DECLARE_WAIT_QUEUE_HEAD(button_queue);
static bool button_pressed = false;
static int press_count = 0;

// 中断处理函数
static irqreturn_t button_isr(int irq, void *dev_id)
{
    press_count++;
    button_pressed = true;

    pr_info("Button pressed! count=%d\n", press_count);

    wake_up_interruptible(&button_queue);

    return IRQ_HANDLED;
}

static __poll_t button_poll(struct file *file, poll_table *wait)
{
    __poll_t mask = 0;

    poll_wait(file, &button_queue, wait);

    if (button_pressed)
        mask |= POLLIN | POLLRDNORM;

    return mask;
}

static ssize_t button_read(struct file *file, char __user *buf,
                           size_t count, loff_t *offset)
{
    int ret;

    if (!button_pressed) {
        if (file->f_flags & O_NONBLOCK)
            return -EAGAIN;
        if (wait_event_interruptible(button_queue, button_pressed))
            return -ERESTARTSYS;
    }

    button_pressed = false;

    ret = put_user(press_count, (int __user *)buf);
    if (ret)
        return ret;

    return sizeof(int);
}

static const struct file_operations fops = {
    .owner = THIS_MODULE,
    .poll = button_poll,
    .read = button_read,
};

static int __init my_init(void)
{
    dev_t dev;
    int ret;

    // 申请GPIO
    ret = gpio_request(gpio_pin, "button");
    if (ret) {
        pr_err("Failed to request GPIO %d\n", gpio_pin);
        return ret;
    }

    // 设置GPIO为输入
    gpio_direction_input(gpio_pin);

    // 获取中断号
    irq_num = gpio_to_irq(gpio_pin);
    pr_info("GPIO %d -> IRQ %d\n", gpio_pin, irq_num);

    // 注册中断处理
    ret = request_irq(irq_num, button_isr,
                      IRQF_TRIGGER_FALLING | IRQF_TRIGGER_RISING,
                      "button_irq", NULL);
    if (ret) {
        pr_err("Failed to request IRQ %d\n", irq_num);
        gpio_free(gpio_pin);
        return ret;
    }

    // 注册字符设备
    ret = alloc_chrdev_region(&dev, 0, 1, DEVICE_NAME);
    if (ret < 0) {
        free_irq(irq_num, NULL);
        gpio_free(gpio_pin);
        return ret;
    }
    my_major = MAJOR(dev);

    cdev_init(&my_cdev, &fops);
    my_cdev.owner = THIS_MODULE;
    cdev_add(&my_cdev, dev, 1);

    my_class = class_create(THIS_MODULE, DEVICE_NAME);
    device_create(my_class, NULL, dev, NULL, DEVICE_NAME);

    pr_info("Button driver loaded: /dev/%s\n", DEVICE_NAME);
    return 0;
}

static void __exit my_exit(void)
{
    device_destroy(my_class, MKDEV(my_major, 0));
    class_destroy(my_class);
    cdev_del(&my_cdev);
    unregister_chrdev_region(MKDEV(my_major, 0), 1);

    free_irq(irq_num, NULL);
    gpio_free(gpio_pin);

    pr_info("Button driver unloaded\n");
}

module_init(my_init);
module_exit(my_exit);
MODULE_LICENSE("GPL");
```

### 4.1 平台驱动

```c
// 文件: exp4_platform/myplatform.c
#include <linux/module.h>
#include <linux/platform_device.h>
#include <linux/mod_devicetable.h>
#include <linux/io.h>
#include <linux/interrupt.h>
#include <linux/of.h>

struct my_device_data {
    void __iomem *base;
    int irq;
    struct resource *mem_res;
};

static irqreturn_t my_irq_handler(int irq, void *dev_id)
{
    struct my_device_data *data = dev_id;

    pr_info("IRQ %d triggered\n", irq);

    // 处理中断
    // ...

    return IRQ_HANDLED;
}

static int my_probe(struct platform_device *pdev)
{
    struct my_device_data *data;
    struct resource *res;
    int irq, ret;

    pr_info("Platform device probed: %s\n", dev_name(&pdev->dev));

    data = devm_kzalloc(&pdev->dev, sizeof(*data), GFP_KERNEL);
    if (!data)
        return -ENOMEM;

    platform_set_drvdata(pdev, data);

    // 获取内存资源
    res = platform_get_resource(pdev, IORESOURCE_MEM, 0);
    if (!res) {
        dev_err(&pdev->dev, "No memory resource\n");
        return -ENODEV;
    }

    // 映射内存
    data->base = devm_ioremap_resource(&pdev->dev, res);
    if (IS_ERR(data->base))
        return PTR_ERR(data->base);

    // 获取中断资源
    irq = platform_get_irq(pdev, 0);
    if (irq < 0) {
        dev_err(&pdev->dev, "No IRQ resource\n");
        return irq;
    }

    // 注册中断
    ret = devm_request_irq(&pdev->dev, irq, my_irq_handler,
                           0, dev_name(&pdev->dev), data);
    if (ret) {
        dev_err(&pdev->dev, "Failed to request IRQ\n");
        return ret;
    }
    data->irq = irq;

    // 获取设备树属性
    if (pdev->dev.of_node) {
        u32 value;
        of_property_read_u32(pdev->dev.of_node, "my-property", &value);
        dev_info(&pdev->dev, "Device tree property: %u\n", value);
    }

    dev_info(&pdev->dev, "Probe successful\n");
    return 0;
}

static int my_remove(struct platform_device *pdev)
{
    struct my_device_data *data = platform_get_drvdata(pdev);

    dev_info(&pdev->dev, "Device removed\n");

    // 清理工作在devm管理下自动完成

    return 0;
}

// 设备树匹配表
static const struct of_device_id my_of_match[] = {
    { .compatible = "vendor,mydevice", },
    { }
};
MODULE_DEVICE_TABLE(of, my_of_match);

// 传统ID匹配表
static const struct platform_device_id my_ids[] = {
    { "mydevice", 0 },
    { }
};
MODULE_DEVICE_TABLE(platform, my_ids);

static struct platform_driver my_driver = {
    .probe = my_probe,
    .remove = my_remove,
    .id_table = my_ids,
    .driver = {
        .name = "myplatform",
        .of_match_table = my_of_match,
    },
};

module_platform_driver(my_driver);
MODULE_LICENSE("GPL");
```
