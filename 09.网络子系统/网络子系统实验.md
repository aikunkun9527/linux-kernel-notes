## 🧪 实验一：网络数据包分析

> [!tip] 目标

### 1.1 使用tcpdump抓包

```bash
# 基本抓包
sudo tcpdump -i eth0

# 指定协议
sudo tcpdump -i eth0 tcp
sudo tcpdump -i eth0 udp
sudo tcpdump -i eth0 icmp

# 指定端口
sudo tcpdump -i eth0 port 80
sudo tcpdump -i eth0 port 22

# 指定主机
sudo tcpdump -i eth0 host 192.168.1.1

# 保存到文件
sudo tcpdump -i eth0 -w capture.pcap

# 读取文件
tcpdump -r capture.pcap

# 详细输出
sudo tcpdump -i eth0 -nn -vvv

# 组合过滤
sudo tcpdump -i eth0 'tcp port 80 and host 192.168.1.1'
```

## 🧪 实验二：Socket编程

> [!tip] 目标
> 编写客户端/服务器程序理解Socket API

### 2.1 TCP服务器

```c
// 文件: exp2_socket/tcp_server.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>

#define PORT 8080
#define BUFFER_SIZE 1024

int main() {
    int server_fd, client_fd;
    struct sockaddr_in address;
    int addrlen = sizeof(address);
    char buffer[BUFFER_SIZE] = {0};
    int opt = 1;

    // 创建socket
    if ((server_fd = socket(AF_INET, SOCK_STREAM, 0)) == 0) {
        perror("socket failed");
        exit(EXIT_FAILURE);
    }

    // 设置SO_REUSEADDR
    if (setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt))) {
        perror("setsockopt failed");
        exit(EXIT_FAILURE);
    }

    address.sin_family = AF_INET;
    address.sin_addr.s_addr = INADDR_ANY;
    address.sin_port = htons(PORT);

    // 绑定
    if (bind(server_fd, (struct sockaddr *)&address, sizeof(address)) < 0) {
        perror("bind failed");
        exit(EXIT_FAILURE);
    }

    // 监听
    if (listen(server_fd, 5) < 0) {
        perror("listen failed");
        exit(EXIT_FAILURE);
    }

    printf("Server listening on port %d...\n", PORT);

    // 接受连接
    if ((client_fd = accept(server_fd, (struct sockaddr *)&address,
                            (socklen_t*)&addrlen)) < 0) {
        perror("accept failed");
        exit(EXIT_FAILURE);
    }

    printf("Connection accepted from %s:%d\n",
           inet_ntoa(address.sin_addr), ntohs(address.sin_port));

    // 接收数据
    int valread = read(client_fd, buffer, BUFFER_SIZE);
    printf("Received: %s\n", buffer);

    // 发送响应
    send(client_fd, "Hello from server", 17, 0);
    printf("Response sent\n");

    close(client_fd);
    close(server_fd);

    return 0;
}
```

### 2.2 TCP客户端

```c
// 文件: exp2_socket/tcp_client.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>

#define PORT 8080
#define BUFFER_SIZE 1024

int main() {
    int sock = 0;
    struct sockaddr_in serv_addr;
    char buffer[BUFFER_SIZE] = {0};

    // 创建socket
    if ((sock = socket(AF_INET, SOCK_STREAM, 0)) < 0) {
        perror("socket creation failed");
        exit(EXIT_FAILURE);
    }

    serv_addr.sin_family = AF_INET;
    serv_addr.sin_port = htons(PORT);

    // 转换IP地址
    if (inet_pton(AF_INET, "127.0.0.1", &serv_addr.sin_addr) <= 0) {
        perror("invalid address");
        exit(EXIT_FAILURE);
    }

    // 连接服务器
    if (connect(sock, (struct sockaddr *)&serv_addr, sizeof(serv_addr)) < 0) {
        perror("connection failed");
        exit(EXIT_FAILURE);
    }

    printf("Connected to server\n");

    // 发送数据
    send(sock, "Hello from client", 17, 0);
    printf("Message sent\n");

    // 接收响应
    read(sock, buffer, BUFFER_SIZE);
    printf("Server response: %s\n", buffer);

    close(sock);

    return 0;
}
```

### 2.3 使用getsockopt/setsockopt

```c
// 文件: exp2_socket/socket_options.c
#include <stdio.h>
#include <sys/socket.h>

int main() {
    int sock = socket(AF_INET, SOCK_STREAM, 0);

    // 获取/设置接收缓冲区大小
    int rcvbuf, optlen = sizeof(rcvbuf);
    getsockopt(sock, SOL_SOCKET, SO_RCVBUF, &rcvbuf, &optlen);
    printf("Default receive buffer: %d\n", rcvbuf);

    // 设置接收缓冲区大小
    int newbuf = 256 * 1024;
    setsockopt(sock, SOL_SOCKET, SO_RCVBUF, &newbuf, sizeof(newbuf));
    getsockopt(sock, SOL_SOCKET, SO_RCVBUF, &rcvbuf, &optlen);
    printf("New receive buffer: %d\n", rcvbuf);

    // 获取发送缓冲区大小
    int sndbuf;
    getsockopt(sock, SOL_SOCKET, SO_SNDBUF, &sndbuf, &optlen);
    printf("Send buffer: %d\n", sndbuf);

    // 获取keepalive设置
    int keepalive;
    getsockopt(sock, SOL_SOCKET, SO_KEEPALIVE, &keepalive, &optlen);
    printf("Keepalive: %s\n", keepalive ? "enabled" : "disabled");

    close(sock);
    return 0;
}
```

## 🧪 实验三：Netfilter模块

> [!tip] 目标
> 编写Netfilter内核模块实现包过滤

### 3.1 简单包过滤模块

```c
// 文件: exp3_netfilter/packet_filter.c
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/netfilter.h>
#include <linux/netfilter_ipv4.h>
#include <linux/ip.h>
#include <linux/tcp.h>
#include <linux/udp.h>
#include <linux/inet.h>

#define BLOCK_PORT 8080

static unsigned int packet_filter_hook(void *priv,
                                       struct sk_buff *skb,
                                       const struct nf_hook_state *state)
{
    struct iphdr *iph;
    struct tcphdr *tcph;
    struct udphdr *udph;
    char saddr[16], daddr[16];

    if (!skb)
        return NF_ACCEPT;

    iph = ip_hdr(skb);
    if (!iph)
        return NF_ACCEPT;

    // 转换IP地址为字符串
    snprintf(saddr, sizeof(saddr), "%pI4", &iph->saddr);
    snprintf(daddr, sizeof(daddr), "%pI4", &iph->daddr);

    switch (iph->protocol) {
    case IPPROTO_TCP:
        tcph = tcp_hdr(skb);
        if (!tcph)
            return NF_ACCEPT;

        pr_info("TCP: %s:%d -> %s:%d\n",
                saddr, ntohs(tcph->source),
                daddr, ntohs(tcph->dest));

        // 阻止特定端口
        if (ntohs(tcph->dest) == BLOCK_PORT) {
            pr_info("Blocking TCP packet to port %d\n", BLOCK_PORT);
            return NF_DROP;
        }
        break;

    case IPPROTO_UDP:
        udph = udp_hdr(skb);
        if (!udph)
            return NF_ACCEPT;

        pr_info("UDP: %s:%d -> %s:%d\n",
                saddr, ntohs(udph->source),
                daddr, ntohs(udph->dest));
        break;

    case IPPROTO_ICMP:
        pr_info("ICMP: %s -> %s\n", saddr, daddr);
        break;

    default:
        pr_info("Other protocol: %d\n", iph->protocol);
        break;
    }

    return NF_ACCEPT;
}

static const struct nf_hook_ops packet_filter_ops = {
    .hook       = packet_filter_hook,
    .pf         = PF_INET,
    .hooknum    = NF_INET_LOCAL_OUT,
    .priority   = NF_IP_PRI_FIRST,
};

static int __init packet_filter_init(void)
{
    int ret;

    ret = nf_register_net_hook(&init_net, &packet_filter_ops);
    if (ret) {
        pr_err("Failed to register netfilter hook\n");
        return ret;
    }

    pr_info("Packet filter module loaded\n");
    return 0;
}

static void __exit packet_filter_exit(void)
{
    nf_unregister_net_hook(&init_net, &packet_filter_ops);
    pr_info("Packet filter module unloaded\n");
}

module_init(packet_filter_init);
module_exit(packet_filter_exit);
MODULE_LICENSE("GPL");
MODULE_AUTHOR("Student");
MODULE_DESCRIPTION("Simple packet filter using Netfilter");
```

### 4.1 虚拟网卡驱动

```c
// 文件: exp4_netdev/virtual_netdev.c
#include <linux/module.h>
#include <linux/netdevice.h>
#include <linux/etherdevice.h>
#include <linux/skbuff.h>

#define DRV_NAME "vnet"

struct vnet_priv {
    struct napi_struct napi;
    struct sk_buff_head rx_queue;
    struct net_device *dev;
};

static int vnet_open(struct net_device *dev)
{
    struct vnet_priv *priv = netdev_priv(dev);

    napi_enable(&priv->napi);
    netif_start_queue(dev);

    pr_info("vnet: device opened\n");
    return 0;
}

static int vnet_stop(struct net_device *dev)
{
    struct vnet_priv *priv = netdev_priv(dev);

    netif_stop_queue(dev);
    napi_disable(&priv->napi);

    pr_info("vnet: device stopped\n");
    return 0;
}

static netdev_tx_t vnet_xmit(struct sk_buff *skb, struct net_device *dev)
{
    struct vnet_priv *priv = netdev_priv(dev);

    // 统计
    dev->stats.tx_packets++;
    dev->stats.tx_bytes += skb->len;

    pr_info("vnet: transmitting packet, len=%d\n", skb->len);

    // 模拟回环：将发送的包放入接收队列
    skb_push(skb, ETH_HLEN);
    __skb_queue_tail(&priv->rx_queue, skb);

    // 调度NAPI接收
    napi_schedule(&priv->napi);

    return NETDEV_TX_OK;
}

static int vnet_poll(struct napi_struct *napi, int budget)
{
    struct vnet_priv *priv = container_of(napi, struct vnet_priv, napi);
    struct sk_buff *skb;
    int work_done = 0;

    while (work_done < budget && (skb = __skb_dequeue(&priv->rx_queue))) {
        skb->protocol = eth_type_trans(skb, priv->dev);
        skb->ip_summed = CHECKSUM_UNNECESSARY;

        priv->dev->stats.rx_packets++;
        priv->dev->stats.rx_bytes += skb->len;

        netif_receive_skb(skb);
        work_done++;
    }

    if (work_done < budget) {
        napi_complete_done(napi, work_done);
    }

    return work_done;
}

static const struct net_device_ops vnet_ops = {
    .ndo_open       = vnet_open,
    .ndo_stop       = vnet_stop,
    .ndo_start_xmit = vnet_xmit,
    .ndo_validate_addr = eth_validate_addr,
};

static void vnet_setup(struct net_device *dev)
{
    struct vnet_priv *priv;

    ether_setup(dev);

    dev->netdev_ops = &vnet_ops;
    dev->flags |= IFF_NOARP;
    dev->features |= NETIF_F_HW_CSUM;

    priv = netdev_priv(dev);
    priv->dev = dev;

    skb_queue_head_init(&priv->rx_queue);
    netif_napi_add(dev, &priv->napi, vnet_poll, 64);

    // 随机MAC地址
    eth_hw_addr_random(dev);
}

static struct net_device *vnet_dev;

static int __init vnet_init(void)
{
    int err;

    vnet_dev = alloc_netdev(sizeof(struct vnet_priv), "vnet%d",
                            NET_NAME_UNKNOWN, vnet_setup);
    if (!vnet_dev)
        return -ENOMEM;

    err = register_netdev(vnet_dev);
    if (err) {
        free_netdev(vnet_dev);
        return err;
    }

    pr_info("vnet: virtual network device registered\n");
    return 0;
}

static void __exit vnet_exit(void)
{
    unregister_netdev(vnet_dev);
    free_netdev(vnet_dev);
    pr_info("vnet: virtual network device unregistered\n");
}

module_init(vnet_init);
module_exit(vnet_exit);
MODULE_LICENSE("GPL");
MODULE_AUTHOR("Student");
MODULE_DESCRIPTION("Virtual network device driver");
```
